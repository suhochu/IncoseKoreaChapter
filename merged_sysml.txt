package ASELCM_SystemOfSystems {
    doc
    /* ASELCM 통합 패키지: LV3(거버넌스)와 LV2(챕터 조직)를 함께 묶는 유일한 위치.
     *
     * 의존 방향: 이 패키지 -> LV3, LV2 (단방향). LV3는 LV2를 import하지 않고, LV2는 이 패키지를 import하지 않는다.
     * 여기서 하는 일:
     *  1) GovernanceSystemContext 와 incoseKoreaChapterContext 를 하나의 상위 part로 포함하고 포트를 connect 한다.
     *  2) chapter 에 행위(perform action)와 수명주기(exhibit state)를 연결한다.
     */

    private import GovernanceSystemAndTooling::GovernanceSystem::*;
    private import ChapterSystemContext::*;
    private import ChapterConOpsAndLifecycle::ConOps::*;
    private import ChapterConOpsAndLifecycle::Lifecycle::*;

    part def AselcmSystemOfSystems {
        doc
        /* System 3(거버넌스)와 System 2(챕터)의 상위 시스템. System 1(LV1 참조모델)은 별도 모델로 유지한다. */

        part governance : GovernanceSystemContext;

        // 구조 baseline(incoseKoreaChapterContext)을 subset 하고 chapter 에 행위/상태만 추가한다.
        // TODO(확인 필요): 패키지 수준 part usage를 ':>' 로 subset 하는 구성은 Syside check 를 통과했으나,
        //   결과 인스턴스가 baseline과 같은 chapter 인스턴스를 공유하는지는 확인하지 못했다.
        //   공유되지 않으면 baseline 쪽 part 를 part def 로 바꾸고 usage 를 이 패키지에서 생성하는 구조로 전환한다.
        part chapterContext :> incoseKoreaChapterContext {
            part :>> chapter {
                // 행위-구조 연결
                // TODO(확인 필요): 각 action 의 in/out 파라미터(internalData 등)는 아직 chapter.repository 와 바인딩하지 않았다.
                perform action sectorReporting : SectorReporting;
                perform action referenceModelDevelopment : ReferenceModelDevelopmentProcess;
                perform action knowledgeExchange : KnowledgeExchangeSession;

                // 수명주기 상태머신: metrics 는 chapter 자신(ChapterMoEsHolder 상속)을 가리킨다.
                exhibit state lifecycle : ChapterLifecycle {
                    in ref :>> metrics = chapter;
                }
            }
        }

        // LV3 정책(out)/지표(in) 포트와 LV2 켤레(~) 포트 연결
        connect governance.governanceInterface to chapterContext.chapter.governanceInterface;
    }

    part aselcmSystemOfSystems : AselcmSystemOfSystems;
} // package ASELCM_SystemOfSystems

             * 각 판정값은 기본값 없이 검토(리뷰/린트) 결과를 입력으로 받는다.
             *
             * 1) Definition(part def, action def, item def, constraint def 등)과 Package는 PascalCase.
             *    숫자와 밑줄(_)은 구분자로만 허용한다. (예: Gate1_WholeSystemScopeRule, SprintMeetings_2026_Q4)

             *    밑줄(_)은 '순번/식별자/날짜' 접미 또는 접두 구분에만 허용한다.
             *    (예: folder1_Meeting, kickOffMeeting_20261009, ai1_setupChannels, need_M001)
             * 3) Enum 값은 lowerCamelCase. 순번이 필요하면 'stage1_...' 처럼 소문자 접두 + 순번 + 밑줄을 쓴다.

             * 6) 주석(doc)으로만 정보를 남기지 않고 정형 속성(attribute)과 참조(ref)로 모델링한다.
             */
            in isPascalCaseDef : Boolean;
            in isPascalCasePackage : Boolean;
            in isCamelCaseUsage : Boolean;
            in isCamelCaseEnumLiteral : Boolean;
            in isUnderscoreUsedOnlyAsSeparator : Boolean;
            isPascalCaseDef and isPascalCasePackage and isCamelCaseUsage and
            isCamelCaseEnumLiteral and isUnderscoreUsedOnlyAsSeparator
            // TODO(확인 필요): 위 판정값을 채우는 자동 점검 수단(Syside Python API 린트 스크립트 등)이 아직 없다.
            //   입력이 바인딩되지 않은 동안에는 판정이 수행되지 않는다(과거의 'default true'로 인한 항상 참 문제는 제거함).

    abstract constraint def AssignmentCheckpointRule {
        doc
        /* 과제 통과 필수 요건(ModelingAssignment.mandatoryCheckpoints)으로 참조되는 모든 Feedback/Gate 규칙의 공통 상위 정의.
         * 새 규칙을 추가할 때는 이 정의를 상속(:>)하면 과제에서 같은 방식으로 참조할 수 있다.
         */
    }

    // ---------------------------------------------------------------------
    // 3.1 학술자문단 핵심 모델링 검증 규칙 (Feedback)
    // ---------------------------------------------------------------------
    constraint def Feedback1_MissionAchievementConOpsRule :> AssignmentCheckpointRule {
        doc
        /* [Feedback 1: 임무 달성 중심 ConOps]
         * 단순 기능 나열이 아닌, 대상시스템(SoI)이 이해관계자의 최상위 임무(Mission)를
         * 달성하는 종단간(End-to-End) 운용 시나리오(ConOps)로 구성되었는가?
         * 운용 개념은 대상시스템의 존재목적(임무 또는 사업목표) 달성을 표현하고 satisfy 관계로 연결해야 한다.
         */
        in isMissionOrientedConOps : Boolean;
        in hasEndToEndScenarioFlow : Boolean;
        isMissionOrientedConOps and hasEndToEndScenarioFlow
    }

    constraint def Feedback2_QuantitativeMOEStructureRule :> AssignmentCheckpointRule {
        doc
        /* [Feedback 2: 정량적 MOE/MOP 체계]
         * 모호한 형용사(빠르게, 안전하게 등)가 배제되고, 이해관계자 효과성 척도(MOE)와
         * 블랙박스 시스템 성능 척도(MOP)가 정량적 수치 및 단위로 구조화되었는가?
         * MOE는 고객/임무/공장 효과 중심의 계층 구조로 정의하고 목표값(Target)/한계값(Threshold)/단위(Unit)를 지정해야 한다.
         */
        attribute minMoeCount : Integer = 1;
        in moeCount : Integer;
        in hasQuantitativeMOEAndMOP : Boolean;
        in isFreeOfVagueAdjectives : Boolean;
        moeCount >= minMoeCount and hasQuantitativeMOEAndMOP and isFreeOfVagueAdjectives
    }

    constraint def Feedback3_BlackBoxAndTimeframeInfraRule :> AssignmentCheckpointRule {
        doc
        /* [Feedback 3: 블랙박스 순수성 및 운용 시점 인프라 정합성]
         * 내부 구현 기술(White-Box) 유출 없이 외부 인프라 및 운용 시점(Timeframe) 전제가
         * 컨텍스트 및 시나리오와 일관되게 연결되었는가?
         * 운용 시점의 PESTEL-CTIMMRA 인프라 가정을 명시하고, 블랙박스 SoI가 외부 환경(조직/물리/자연)과
         * 상호작용하는 것에 초점을 두고 시나리오를 작성해야 한다.
         */
        in isBlackBoxContextConsistent : Boolean;
        in isTimeframeInfraLinked : Boolean;
        isBlackBoxContextConsistent and isTimeframeInfraLinked
    }

    // ---------------------------------------------------------------------
    // 3.2 과제 산출물 자체 감사 게이트 (Verification Gates)
    // ---------------------------------------------------------------------
    constraint def Gate1_WholeSystemScopeRule :> AssignmentCheckpointRule {
        doc
        /* [Gate 1: Whole-System]
         * 대상시스템(SoI)이 하위 모듈/센서/단일 제어기가 아닌, 독립적으로 이해관계자에게
         * 임무를 완수할 수 있는 최상위 전체 시스템(Whole-System)으로 선정되었는가?
         */
        in isIndependentWholeSystem : Boolean;
        isIndependentWholeSystem
    }

    constraint def Gate2_BlackBoxSolutionFreeRule :> AssignmentCheckpointRule {
        doc
        /* [Gate 2: Black-Box (Solution-Free 원칙)]
         * 내부 아키텍처(특정 MCU, ECU, SW 함수, 내부 통신 버스 규격, 알고리즘)가 철저히 배제되고
         * 경계면의 입출력(Port/Item)만 노출되었는가?
         */
        in hasZeroInternalComponentsExposed : Boolean;
        hasZeroInternalComponentsExposed
    }

    constraint def Gate3_TimeframeAssumptionLinkRule :> AssignmentCheckpointRule {
        doc
        /* [Gate 3: 시점 전제 연계]
         * 단순 연도 텍스트 기재를 넘어, 운용 시점(예: 2030~2040년)의 전제 조건
         * (①외부 인프라 성숙도, ②법규/제도, ③위협/외란 수준, ④수명주기 진화 특성)이 반영되었는가?
         */
        in hasFourCategoryTimeframeAssumptions : Boolean;
        hasFourCategoryTimeframeAssumptions
    }

    constraint def Gate4_ODDSpatialGroupingRule :> AssignmentCheckpointRule {
        doc
        /* [Gate 4: 장소 그룹핑]
         * 무분별한 개별 지명 나열 대신, 물리적·기능적 부하(Stress)의 동질성을 기준으로
         * 그룹 A(공칭/정규), 그룹 B(복합/고부하), 그룹 C(극한/기능저하)로 추상화되었는가?
         */
        in hasThreeTierODDGrouping : Boolean;
        hasThreeTierODDGrouping
    }

    constraint def Gate5_ThreeExternalScopesCompletenessRule :> AssignmentCheckpointRule {
        doc
        /* [Gate 5: 외부 범위 완전성 및 외란 아이템화]
         * 외부 조직(Org), 외부 물리 시스템(Physical), 외부 자연 환경(Nature)이 각각 최소 개수 이상 식별되었으며,
         * 자연 환경 외란이 문학적 묘사가 아닌 EnvironmentalDisturbanceItem으로 정형화되었는가?
         * 최소 개수는 minActorCountPerScope 한 곳에서만 정의한다.
         */
        attribute minActorCountPerScope : Integer = 2;
        in externalOrgCount : Integer;
        in externalPhysicalCount : Integer;
        in externalNatureCount : Integer;
        in isNaturalDisturbanceItemized : Boolean;

        externalOrgCount >= minActorCountPerScope and
        externalPhysicalCount >= minActorCountPerScope and
        externalNatureCount >= minActorCountPerScope and
        isNaturalDisturbanceItemized
    }

    constraint def Gate6_NoIsolatedActorTraceabilityRule :> AssignmentCheckpointRule {
        doc
        /* [Gate 6: 고립 액터 방지 (Traceability 강제)]
         * 컨텍스트에 정의된 모든 외부 액터(조직/물리/자연)와 시점·장소 그룹이
         * 최종 5-Tuple 운용 시나리오(공칭 + 외란 적응 + 비상 폴백/MRC)에 최소 1회 이상 등장하는가?
         */
        in hasZeroIsolatedExternalActors : Boolean;
        in includesNominalDisturbanceAndMRCFlows : Boolean;
        hasZeroIsolatedExternalActors and includesNominalDisturbanceAndMRCFlows
    }

    // ---------------------------------------------------------------------
    // 3.3 LV1 참조모델 과제 통합 승인 게이트 (Feedback + Gate 통합)
    // ---------------------------------------------------------------------
    attribute def AssignmentGateChecklist {
        doc
        /* 통합 승인 게이트(ReferenceModelAssignmentApprovalGate)의 입력 전체를 속성별로 분리해 담는 체크리스트.
         * 속성 이름은 각 Feedback/Gate constraint def의 파라미터 이름과 같다.
         * 각 항목은 개별 속성이므로 하나의 판정값을 여러 파라미터에 중복 입력하는 정보 손실이 없다.
         */
        // Feedback 1
        attribute isMissionOrientedConOps : Boolean default false;
        attribute hasEndToEndScenarioFlow : Boolean default false;
        // Feedback 2
        attribute moeCount : Integer default 0;
        attribute hasQuantitativeMOEAndMOP : Boolean default false;
        attribute isFreeOfVagueAdjectives : Boolean default false;
        // Feedback 3
        attribute isBlackBoxContextConsistent : Boolean default false;
        attribute isTimeframeInfraLinked : Boolean default false;
        // Gate 1 ~ 4
        attribute isIndependentWholeSystem : Boolean default false;
        attribute hasZeroInternalComponentsExposed : Boolean default false;
        attribute hasFourCategoryTimeframeAssumptions : Boolean default false;
        attribute hasThreeTierODDGrouping : Boolean default false;
        // Gate 5
        attribute externalOrgCount : Integer default 0;
        attribute externalPhysicalCount : Integer default 0;
        attribute externalNatureCount : Integer default 0;
        attribute isNaturalDisturbanceItemized : Boolean default false;
        // Gate 6
        attribute hasZeroIsolatedExternalActors : Boolean default false;
        attribute includesNominalDisturbanceAndMRCFlows : Boolean default false;
    }

    constraint def ReferenceModelAssignmentApprovalGate :> AssignmentCheckpointRule {
        doc
        /* 워킹그룹이 제출한 LV1 모델이 KnowledgeRepository에 정식 Baseline으로 승격되기 위한 통합 품질 게이트.
         * 입력은 checklist 하나이며, 판정 기준 설명은 각 Feedback/Gate constraint def의 doc에만 둔다.
         */
        in checklist : AssignmentGateChecklist;

        constraint fb1 : Feedback1_MissionAchievementConOpsRule {
            in :>> isMissionOrientedConOps = checklist.isMissionOrientedConOps;
            in :>> hasEndToEndScenarioFlow = checklist.hasEndToEndScenarioFlow;
        }
        constraint fb2 : Feedback2_QuantitativeMOEStructureRule {
            in :>> moeCount = checklist.moeCount;
            in :>> hasQuantitativeMOEAndMOP = checklist.hasQuantitativeMOEAndMOP;
            in :>> isFreeOfVagueAdjectives = checklist.isFreeOfVagueAdjectives;
        }
        constraint fb3 : Feedback3_BlackBoxAndTimeframeInfraRule {
            in :>> isBlackBoxContextConsistent = checklist.isBlackBoxContextConsistent;
            in :>> isTimeframeInfraLinked = checklist.isTimeframeInfraLinked;
        }

        constraint gate1 : Gate1_WholeSystemScopeRule {
            in :>> isIndependentWholeSystem = checklist.isIndependentWholeSystem;
        }
        constraint gate2 : Gate2_BlackBoxSolutionFreeRule {
            in :>> hasZeroInternalComponentsExposed = checklist.hasZeroInternalComponentsExposed;
        }
        constraint gate3 : Gate3_TimeframeAssumptionLinkRule {
            in :>> hasFourCategoryTimeframeAssumptions = checklist.hasFourCategoryTimeframeAssumptions;
        }
        constraint gate4 : Gate4_ODDSpatialGroupingRule {
            in :>> hasThreeTierODDGrouping = checklist.hasThreeTierODDGrouping;
        }
        constraint gate5 : Gate5_ThreeExternalScopesCompletenessRule {
            in :>> externalOrgCount = checklist.externalOrgCount;
            in :>> externalPhysicalCount = checklist.externalPhysicalCount;
            in :>> externalNatureCount = checklist.externalNatureCount;
            in :>> isNaturalDisturbanceItemized = checklist.isNaturalDisturbanceItemized;
        }
        constraint gate6 : Gate6_NoIsolatedActorTraceabilityRule {
            in :>> hasZeroIsolatedExternalActors = checklist.hasZeroIsolatedExternalActors;
            in :>> includesNominalDisturbanceAndMRCFlows =
                checklist.includesNominalDisturbanceAndMRCFlows;
        }

        // 통합 판정식: 하위 constraint usage 자체를 AND 연산으로 평가
        fb1 and fb2 and fb3 and gate1 and gate2 and gate3 and gate4 and gate5 and gate6
    }

    // ---------------------------------------------------------------------
    // 3.4 모델 검수 및 승인 프로세스 (WG Lead → 학술자문단 → 챕터 리더십)
    // ---------------------------------------------------------------------
    action def ModelReviewProcess {
        doc
        /* 정기적인 챕터 모델 및 LV1 과제 산출물 검수·승인 워크플로우.
         * 검토 순서는 ModelingConventions::VersionControlPolicy(WG Lead → 학술자문단 → 보안/형상 승인)와 같다.
         * 게이트 미통과 또는 자문단 미승인이면 reviseModel(SubmissionStatusKind::revisionRequired에 해당)을 거쳐 재제출한다.
         */
        in attribute submittedChecklist : AssignmentGateChecklist;
        in attribute isAdvisorApproved : Boolean default false;

        constraint gateEvaluation : ReferenceModelAssignmentApprovalGate {
            in :>> checklist = submittedChecklist;
        }

        action submitModelUpdate {
            doc
            /* 워킹그룹 또는 회원이 스캐폴드 기반 모델 제출 */
        }
        action checkAutomatedConventionsAndGates {
            doc
            /* 가이드 Gate 및 스캐폴드 상속·포트 대칭성 1차 점검 */
        }
        action reviewByWGLead {
            doc
            /* 워킹그룹 리드(WG Lead)의 1차 모델 검토 */
        }
        action reviewByAcademicAdvisors {
            doc
            /* 학술자문단의 피드백 1·2·3 충족 여부 심사 */
        }
        action reviewByChapterLeadership {
            doc
            /* 챕터 대표 및 총무의 보안 규정(사내 데이터 배제) 확인 및 최종 승인 */
        }
        action mergeAndDeploy {
            doc
            /* GitHub 메인 브랜치 및 Google Workspace 단일 진실 공급원(SSOT) 배포 */
        }
        action reviseModel {
            doc
            /* 게이트 미통과 또는 자문단 지적 사항을 반영하는 재작업 */
        }

        first start;
        then submitModelUpdate;
        then checkAutomatedConventionsAndGates;
        then gateDecision;

        // TODO(확인 필요): decide/if/else 분기 문법은 Syside 0.10.x 에서 check 통과를 확인했으나,
        //   gateEvaluation.result 를 가드로 쓰는 실행 의미(시뮬레이션)는 확인하지 못했다.
        decide gateDecision;
            if gateEvaluation.result then reviewByWGLead;
            else reviseModel;

        first reviewByWGLead then reviewByAcademicAdvisors;
        first reviewByAcademicAdvisors then advisorDecision;

        decide advisorDecision;
            if isAdvisorApproved then reviewByChapterLeadership;
            else reviseModel;

        first reviewByChapterLeadership then mergeAndDeploy;
        first mergeAndDeploy then done;
        first reviseModel then submitModelUpdate;
    }

    // ---------------------------------------------------------------------
    // 3.5 System 2 (챕터 조직) 성장 단계 전환 조건 (Maturity Gate)
    // ---------------------------------------------------------------------
    attribute def MaturityGateThresholds {
        doc
        /* Phase 전환 조건의 임계값 묶음. 모든 값은 '이상(>=)' 최소값이다. */
        attribute presentations : Integer;
        attribute refModels : Integer;
        attribute aiSkills : Integer;
        attribute activeMembers : Integer;
        attribute sepMembers : Integer;
    }

    // Phase 임계값의 단일 정의 위치. 판정식과 doc은 숫자를 다시 적지 않고 이 값을 참조한다.
    attribute phase1To2Thresholds : MaturityGateThresholds {
        :>> presentations = 10;
        :>> refModels = 3;
        :>> aiSkills = 3;
        :>> activeMembers = 16; // 정원 '15명 초과'를 정수 최소값으로 표현
        :>> sepMembers = 3;
    }

    attribute phase2To3Thresholds : MaturityGateThresholds {
        :>> presentations = 0; // Phase 2 -> 3 판정식에서는 사용하지 않음
        :>> refModels = 3;
        :>> aiSkills = 5;
        :>> activeMembers = 30;
        :>> sepMembers = 10;
    }

    abstract constraint def PhaseTransitionGateCondition {
        doc
        /* 조직 성장 조건 (Maturity Gate):
         * LV1 참조모델 승인 누적 건수 및 세미나 발표 건수 등 정량적 지표를 기반으로
         * System 2(KoreaChapter)의 Lifecycle Phase 전환 가능 여부를 판정하는 공통 입력 정의.
         * 구체 판정식과 임계값은 하위 constraint def(Phase1ToPhase2GateConstraint 등)와
         * MaturityGateThresholds 값에만 존재한다.
         */
        in presentations : Integer;
        in refModels : Integer;
        in aiSkills : Integer;
        in activeMembers : Integer;
        in sepMembers : Integer;
        in thresholds : MaturityGateThresholds;
    }

    // ---- Phase 1 -> Phase 2 전환 게이트 (온라인 스터디 -> 오프라인 세미나 정례화) ----
    constraint def Phase1ToPhase2GateConstraint :> PhaseTransitionGateCondition {
        doc
        /* [Phase 1 -> Phase 2 전환 조건 (기업 스폰서 조건 제외)]
         * 판정식: (발표/사례 자료 OR 고정 활동 인원 중 하나) AND 참조모델 AND AI Skills AND SEP 보유 인원.
         * 임계값은 phase1To2Thresholds 를 참조한다.
         */
        in :>> thresholds : MaturityGateThresholds default phase1To2Thresholds;

        (presentations >= thresholds.presentations or activeMembers >= thresholds.activeMembers) and
        refModels >= thresholds.refModels and
        aiSkills >= thresholds.aiSkills and
        sepMembers >= thresholds.sepMembers
    }

    // ---- Phase 2 -> Phase 3 전환 게이트 (세미나 정례화 -> 공식 WG 및 학회 Joint WG) ----
    constraint def Phase2ToPhase3GateConstraint :> PhaseTransitionGateCondition {
        doc
        /* [Phase 2 -> Phase 3 전환 조건 (기업 스폰서 조건 제외)]
         * 판정식: 고정 활동 인원 AND SEP 자격 보유자 AND AI Skills AND 참조모델 유지.
         * 임계값은 phase2To3Thresholds 를 참조한다.
         */
        in :>> thresholds : MaturityGateThresholds default phase2To3Thresholds;

        activeMembers >= thresholds.activeMembers and
        sepMembers >= thresholds.sepMembers and
        aiSkills >= thresholds.aiSkills and
        refModels >= thresholds.refModels
    }
} // package QualityGatesAndReviewProcess

        subject chapter : KoreaChapter;
        attribute isSolutionFree : Boolean;

        require constraint problemSpaceSeparated { isSolutionFree }

    // ================================================
    requirement def ChapterSystemIdea :> Objective {

        /* 1. "현장의 문제를 직접 해결하는 시스템 엔지니어링 실전 랩(Lab), INCOSE Korea 'Do-Tank' 챕터".

        attribute coversOperationalRequirementToCodeGeneration : Boolean;

    requirement chapterObjective : ChapterObjective;

package ChapterMetricsAndGuards {
    doc
    /* [MagicGrid: Measures of Effectiveness (MoEs) & System Parameters]
     * 기업 스폰서 등 외부 환경 변수를 배제하고, 챕터 내부의 산출물과 인적 역량만으로
     * 성장을 측정 및 통제하는 핵심 성과 지표(Core MoEs)와 Phase 전환 판정 결과.
     *
     * 판정식과 임계값은 LV3(QualityGatesAndReviewProcess)가 한 곳에서 정의한다.
     * 이 패키지는 지표 값을 그 constraint의 입력에 연결하고 판정 결과를 파생할 뿐, 숫자를 다시 적지 않는다.
     */

    private import ScalarValues::*;
    private import QualityGatesAndReviewProcess::*;

    part def ChapterMoEsHolder {
        doc
        /* KoreaChapter(SoI)가 상속받는 핵심 정량 지표(MoEs).
         * 각 지표의 목표 임계값은 QualityGatesAndReviewProcess::phase1To2Thresholds / phase2To3Thresholds 를 참조한다.
         */

        attribute accumulatedPresentationsCount : Integer; // MoE: 누적 발표/사례 자료 수 (수동 입력)
        attribute completedReferenceModelsCount : Integer; // MoE: 완성된 도메인 참조모델 수 (KoreaChapter에서 repository 크기로 파생)
        attribute developedAISkillsCount : Integer; // MoE: 단계별 AI Skills 수 (KoreaChapter에서 repository 크기로 파생)
        attribute regularActiveMembersCount : Integer; // MoE: 고정 활동 인원 수 (수동 입력)
        attribute sepCertifiedMembersCount : Integer; // MoE: SEP 자격 보유자 수 (수동 입력)

        // Phase 1 -> Phase 2 전환 판정: 지표 값을 LV3 constraint의 입력에 연결
        constraint phase1To2GateCheck : Phase1ToPhase2GateConstraint {
            in :>> presentations = accumulatedPresentationsCount;
            in :>> refModels = completedReferenceModelsCount;
            in :>> aiSkills = developedAISkillsCount;
            in :>> activeMembers = regularActiveMembersCount;
            in :>> sepMembers = sepCertifiedMembersCount;
        }

        // Phase 2 -> Phase 3 전환 판정
        constraint phase2To3GateCheck : Phase2ToPhase3GateConstraint {
            in :>> presentations = accumulatedPresentationsCount;
            in :>> refModels = completedReferenceModelsCount;
            in :>> aiSkills = developedAISkillsCount;
            in :>> activeMembers = regularActiveMembersCount;
            in :>> sepMembers = sepCertifiedMembersCount;
        }

        // Phase 전환 Guard용 파생 Boolean (상태머신 transition의 accept when 조건에서 참조).
        // 판정식은 constraint 한 곳에만 있고, 여기서는 그 결과(result)만 가져온다.
        // TODO(확인 필요): constraint usage 자체는 Boolean에 type-conform 하지 않아(Syside type-error) '.result'로 참조했다.
        //   Syside 정적 평가기는 constraint usage를 평가하지 못하고('no result expression') '.result'도 값이 아닌
        //   result feature 참조로 돌려준다. 실행(시뮬레이션) 수준에서의 평가 확인이 필요하다.
        //   대안: 파생 Boolean만 남기고 constraint는 'assert constraint'로 사용(판정식 1곳 유지).
        attribute isPhase1To2Ready : Boolean = phase1To2GateCheck.result;
        attribute isPhase2To3Ready : Boolean = phase2To3GateCheck.result;
    }
} // package ChapterMetricsAndGuards

        open; // 미착수
        inProgress; // 진행 중
        done; // 완료
    }

    item def Sprint {
        doc
        /* 스프린트 단위. Action Item과 과제는 문자열("Sprint 2") 대신 이 요소를 ref로 참조한다.
         * 새 스프린트는 분기별 회의록 패키지(SprintMeetings_YYYY_Qn)에 usage로 추가한다.
         */
        attribute sprintNo : Integer;
        attribute endDate : Iso8601DateTimeStructure; // 스프린트 마감일
    }

    action def ChapterSprintMeeting :> KnowledgeExchangeSession {
        attribute meetingDate : Iso8601DateTimeStructure;
        attribute location : String;
        ref attendees : IncoseKoreaChapterMember [1..*];
        attribute keyDecisions : String [0..*];

        // 회의록 산출물은 항상 '1. Meeting' 폴더에 저장된다(경로 문자열은 folder.relativePath 에서만 얻는다).
        out item :>> producedArtifact : ChapterArtifact {
            :>> category = ArtifactCategoryKind::meetingMinutes;
            ref :>> folder = incoseKoreaChapterContext.chapter.repository.folder1_Meeting;
        }
    }

    action def ActionItemTask {
        ref owner : IncoseKoreaChapterMember [1..*];
        attribute taskDescription : String;
        ref dueSprint : Sprint [0..1]; // 스프린트 마감이 기한이면 이 ref만 지정
        attribute dueDate : Iso8601DateTimeStructure [0..1] default dueSprint.endDate; // 스프린트와 무관한 기한만 재정의
        attribute isImmediate : Boolean default false; // 즉시 처리 항목이면 true (dueDate 없음)
        attribute completionStatus : TaskCompletionKind default TaskCompletionKind::open;
    }

    // =========================================================================
    // 2. SE 모델링 과제(Assignment), 학술자문단 피드백 및 제출물 관리 스키마
    // =========================================================================
    enum def SubmissionStatusKind {
        notStarted; // 착수 전
        inProgress; // 워킹그룹 모델링 진행 중
        underAcademicReview; // 학술자문단 리뷰 및 게이트 검증 중
        revisionRequired; // 학술자문단 피드백 반영 및 보완 필요

    }

        /* 학술자문단이 과제 및 모델에 대해 제시한 피드백과 해소 상태.
         * 피드백이 특정 회의에서 나왔다면 sourceMeeting만 지정하면 feedbackDate가 회의 일시에서 파생된다.
         */
        ref reviewers : IncoseKoreaChapterMember [1..*];
        ref action sourceMeeting : ChapterSprintMeeting [0..1];
        attribute feedbackDate : Iso8601DateTimeStructure [0..1] default sourceMeeting.meetingDate;
        attribute feedbackPoints : String [1..*];
        attribute isResolvedInModel : Boolean default false;
    }

    // SE 모델링 과제 마스터 정의 (ActionItemTask 확장)
    action def ModelingAssignment :> ActionItemTask {
        doc
        /* 각 스프린트 주기에 부여되는 시스템 엔지니어링 모델링 과제 템플릿.
         * 실무 가이드 문서, 필수 반영 체크포인트(Gate/Feedback constraint), 학술자문단 피드백을 통합 관리함.
         * 체크포인트 설명은 각 constraint def의 doc에만 있으며 여기서는 참조(ref)만 한다.
         */
        attribute assignmentNo : Integer [1..*];
        attribute stage : SEAssignmentStageKind [1..*];
        ref guideArtifact : ChapterArtifact [0..1]; // 참고 실무 가이드(PDF 등)
        attribute scaffoldPackageName : String [0..1]; // 상속해야 할 LV1 공통 스캐폴드 패키지명
        ref constraint mandatoryCheckpoints : AssignmentCheckpointRule [1..*]; // 과제 통과 필수 요건
        item advisoryFeedbacks : AdvisoryFeedback [0..*]; // 학술자문단 피드백 내역
        attribute startDate : Iso8601DateTimeStructure;
    }

    // 워킹그룹별 과제 제출물 및 품질 게이트 심사 결과 트래커
    item def AssignmentSubmission {
        doc
        /* 각 워킹그룹이 특정 과제(ModelingAssignment)에 대해 제출한 LV1 참조모델과
         * LV3 거버넌스 품질 게이트(학술자문단 피드백 + 실무 가이드 Gate) 통과 현황을 관리함.
         * 도메인, SoI, 저장 폴더, 소유자는 workingGroup 에서 파생하므로 제출 usage에는 다시 적지 않는다.
         */
        ref targetAssignment : ModelingAssignment;
        ref workingGroup : WorkingGroup;
        // 기본값은 WG 참여자. 제출자가 WG 참여자와 다를 때만 usage에서 재정의한다.
        ref submitters : IncoseKoreaChapterMember [1..*] default workingGroup.participants;
        attribute status : SubmissionStatusKind default SubmissionStatusKind::inProgress;

        // 게이트 입력 전체(Feedback + Gate 항목)를 하나의 체크리스트로 보관
        attribute checklist : AssignmentGateChecklist;

        // TODO(확인 필요): 속성 이름(checklist)이 constraint 파라미터 이름과 같아 'AssignmentSubmission::checklist' 로 한정했다.
        //   Syside Python API로 이 바인딩의 referent 가 AssignmentSubmission::checklist (자기 자신이 아님)임을 확인했다.
        //   실행 시 소유 인스턴스의 checklist로 해석되는지는 시뮬레이션 수준에서 추가 확인이 필요하다.
        constraint gateCheck : ReferenceModelAssignmentApprovalGate {
            in :>> checklist = AssignmentSubmission::checklist;
        }

        // TableView 표시용 파생 Boolean. 판정식은 gateCheck 한 곳에만 있다.
        // TODO(확인 필요): constraint usage는 Boolean에 직접 type-conform 하지 않아 '.result'로 참조했다(실행 의미 확인 필요).
        attribute isGatePassed : Boolean = gateCheck.result;

        item submittedModel : ReferenceModelArtifact {
            :>> domain = workingGroup.domain;
            :>> targetSoIName = workingGroup.targetSystemScope;
            :>> modelFilePath = workingGroup.targetSysmlPackagePath;
            ref :>> folder = workingGroup.targetFolder;
            ref :>> owner = workingGroup.participants;
            :>> isGateApproved = isGatePassed;
        }

        // 상태와 게이트 결과의 모순 방지: 승인된 Baseline이면 게이트를 반드시 통과해야 한다.
        assert constraint statusGateConsistency {
            status == SubmissionStatusKind::approvedBaseline implies isGatePassed
        }
    }

##2_ASELCM LV2_Chapter_Organization/5_Views_and_Viewpoints/1_StakeholdersConcerns.sysml##
package StakeholdersConcerns {
    doc
    /* [이해관계자 관심사(Concerns) 정의]
     * Stakeholders, ActorAndInterfaces, SystemContext 구조를 기반으로
     * 핵심 관심사를 정의하고, 이를 Viewpoints에 빈틈없이 프레임(frame)한다.
     * 공통 subject(theChapter)와 stakeholder(member)는 KoreanChapterConcern에 한 번만 두고,
     * 파생 Concern에는 추가되는 stakeholder만 남긴다.
     *
     * 1. 'administrative operations viewpoint' 매핑 관심사:
     *    - OperationalFlowConcern (운영 흐름 및 정기 보고)
     *    - KnowledgeAssetConcern (지식 저장소 및 참조모델/아티팩트 형상 관리)
     *    - LifecycleEvolutionConcern (챕터 성장 단계 전환 및 핵심 MoE 관리)
     * 2. 'member value viewpoint' 매핑 관심사:
     *    - StakeholderIdentificationConcern (회원 도메인·전문성·기여 자원 식별)
     *    - ValueRationaleValidationConcern (회원 핵심 가치 What, 도출 근거 Why, 검증 기준 Validation)
     *    - RiskSecurityConcern (기업 IP 보호 및 데이터 익명화 보안 규칙)
     * 3. 'incose alignment viewpoint' 매핑 관심사:
     *    - BoundaryDependencyConcern (INCOSE 본부/섹터/TechOps 및 툴 벤더 외부 의존성)
     * 4. 'peer society connection viewpoint' 매핑 관심사:
     *    - CapabilityValueConcern (국내 타 학회 KOSSE/KCOSE 대비 고유 역량 및 상호보완성)
     * 5. 'governance and quality viewpoint' 매핑 관심사:
     *    - GovernanceQualityConcern (LV3 거버넌스 규약, 품질 게이트 및 승인 상태)
     */

    private import StakeholderDefinitions::*;
    private import ExternalActorsAndInterfaces::*;
    private import ChapterSystemContext::*;
    private import GovernanceSystemAndTooling::GovernanceSystem::*;

    concern def KoreanChapterConcern {
        subject theChapter : KoreaChapter;
        stakeholder member : Member;
    }

    // -------------------------------------------------------------------------
    // [Group 1] Member Value Viewpoint 매핑 관심사
    // -------------------------------------------------------------------------
    concern def StakeholderIdentificationConcern :> KoreanChapterConcern {
        require constraint {
            doc
            /* 시스템(챕터)과 직접 상호작용하는 회원 액터들의 소속, 도메인, 지역, 가용 기여 자원 식별. */
        }
    }

    concern def ValueRationaleValidationConcern :> KoreanChapterConcern {
        require constraint {
            doc
            /* 시스템이 이해관계자에게 제공해야 할 핵심 가치(What), 도출
             * 근거(Why), 타당성 확인 기준(Validation), 상충 시 조정
             * 근거(Priority).
             */
        }
    }

    concern def RiskSecurityConcern :> KoreanChapterConcern {
        require constraint {
            doc
            /* 회원들이 소속 기업의 지적 재산(IP)이나 민감한 기술 정보를
             * 침해하지 않으면서도 지식을 교류할 수 있는 안전한 환경
             * (Boundary) 및 보안 규칙(SecurityRule)은 어떻게 구축하는가.
             */
        }
    }

    // -------------------------------------------------------------------------
    // [Group 2] Administrative Operations Viewpoint 매핑 관심사
    // -------------------------------------------------------------------------
    concern def OperationalFlowConcern :> KoreanChapterConcern {
        subject :>> theChapter; // Syside: stakeholder를 추가하는 Concern은 subject 재선언이 첫 파라미터여야 함
        stakeholder sector : ContextINCOSESector;
        require constraint {
            doc
            /* 시스템이 요구사항을 충족하기 위해 운영 기간 동안 어떻게
             * 동작(스프린트, 과제 수행, 세미나)하고 정보를 교환하는가.
             */
        }
    }

    concern def KnowledgeAssetConcern :> KoreanChapterConcern {
        require constraint {
            doc
            /* 챕터 내에서 생성되는 지식(회의록, 가이드, LV1 참조모델 코드)은 어떤 구조를
             * 가지며, 단일 진실 공급원(KnowledgeRepository)에 어떻게 저장되고 추적되는가.
             */
        }
    }

    concern def LifecycleEvolutionConcern :> KoreanChapterConcern {
        require constraint {
            doc
            /* 시스템(챕터)이 시간에 따라 어떻게 상태(Phase 1 -> Phase 2 -> Phase 3)가 변하고
             * 핵심 성과 지표(Core MoEs)를 통해 진화하는가.
             */
        }
    }

    // -------------------------------------------------------------------------
    // [Group 3] INCOSE Alignment Viewpoint 매핑 관심사
    // -------------------------------------------------------------------------
    concern def BoundaryDependencyConcern :> KoreanChapterConcern {
        subject :>> theChapter;
        stakeholder vendor : ContextToolVendor;
        stakeholder techOps : ContextTechnicalOperations;
        require constraint {
            doc
            /* 시스템의 경계(Boundary), 외부 의존성(Dependencies), 가용
             * 툴 라이선스 및 INCOSE 글로벌 리소스(SE Handbook 등) 연계 한계.
             */
        }
    }

    // -------------------------------------------------------------------------
    // [Group 4] Peer Society Connection Viewpoint 매핑 관심사
    // -------------------------------------------------------------------------
    concern def CapabilityValueConcern :> KoreanChapterConcern {
        subject :>> theChapter;
        stakeholder peerMember : PeerSocietyMember;
        stakeholder peerOrg : ContextPeerSocietyOrganization;
        require constraint {
            doc
            /* 이 시스템이 궁극적으로 한국의 시스템 엔지니어링 생태계
             * (상위 시스템)에 타 학회(KOSSE/KCOSE) 대비 어떤 고유한 역량(Capability)을 제공하는가.
             */
        }
    }

    // -------------------------------------------------------------------------
    // [Group 5] Governance and Quality Viewpoint 매핑 관심사
    // -------------------------------------------------------------------------
    concern def GovernanceQualityConcern :> KoreanChapterConcern {
        subject :>> theChapter;
        stakeholder governance : GovernanceSystemContext;
        require constraint {
            doc
            /* LV3 거버넌스가 하달하는 모델링 규약과 품질 게이트(학술자문단 피드백, 가이드 Gate)를
             * 워킹그룹 제출물이 충족하는지, 그리고 승인 상태(status)가 게이트 판정 결과와
             * 모순되지 않는지를 어떻게 추적하는가.
             */
        }
    }
} // package StakeholdersConcerns

package ChapterViewpoints {

    viewpoint 'administrative operations viewpoint' {
        // 행정/운영 관점이 다루는 관심사(운영 흐름 + 지식자산/참조모델 관리 + 수명주기/MoE 진화)를 모두 frame
        frame concern operationalFlow : OperationalFlowConcern;
        frame concern knowledgeAsset : KnowledgeAssetConcern;
        frame concern lifecycleEvolution : LifecycleEvolutionConcern;

        require constraint {
            doc
            /* 1. Chapter의 효과적이고 지속적인 운영을 위한 행정적인 관점
             * - 운영 흐름 및 과제 스프린트 프로세스 (OperationalFlowConcern)
             * - 지식 저장소(KnowledgeRepository) 및 LV1 참조모델 형상 관리 (KnowledgeAssetConcern)
             * - 향후 챕터 성장 단계(Phase 1->2->3) 및 핵심 MoE 전환 규칙 (LifecycleEvolutionConcern)
             */
        }
    }

    viewpoint 'member value viewpoint' {
        // 회원 가치 관점이 다루는 관심사(회원 식별 + 핵심 가치/근거/검증 + 기업 보안 규칙)를 모두 frame
        frame concern stakeholderId : StakeholderIdentificationConcern;
        frame concern valueRationale : ValueRationaleValidationConcern;
        frame concern riskSecurity : RiskSecurityConcern;

        require constraint {
            doc
            /* 2. Chapter가 각 멤버들에게 실질적인 긍정적인 영향을 주기 위한 관점
             * - 회원들의 도메인·지역·가용 기여 자원 식별 (StakeholderIdentificationConcern)
             * - 회원별 요구사항(What), 근거(Why), 검증 기준(Validation) 충족 (ValueRationaleValidationConcern)
             * - 개인 및 소속 기업의 정보보안 제약 조건(SecurityRules) 준수 (RiskSecurityConcern)

        }
    }

        frame concern boundaryDependency : BoundaryDependencyConcern;

        require constraint {
            doc

             */
        }
    }

    viewpoint 'peer society connection viewpoint' {
        frame concern capabilityValue : CapabilityValueConcern;

        require constraint {
            doc
            /* 4. 기존 시스템 엔지니어링 모임(KCOSE, KOSSE 등)과의 연계 관점
             * - 국내 타 학회와의 상호 보완적 역할, 챕터 대 챕터 차원의 정보 교환,

        }
    }

    viewpoint 'governance and quality viewpoint' {

            doc
            /* 5. LV3 거버넌스와 품질 보증 관점

             */
        }
    }

package LV1_BlackBox_ConOps_Scaffold {
    doc
    /* [InKC LV1 공통 스캐폴드 패키지]
     * 각 도메인 워킹그룹(자동차, 우주, 스마트팩토리, 반도체장비 등)이 상속(:>)하여
     * 대상시스템(SoI)의 블랙박스 정의와 운용개념(ConOps)을 일관되게 구축하기 위한 추상 메타모델.
     *
     * [핵심 설계 원칙]
     * 1. 존재 목적 달성 표현 (피드백 1): ConOps 시나리오가 임무/사업 목표(MissionRequirement)를 satisfy함.
     * 2. 정량적 MOE 구조화 (피드백 2): 임무 달성 여부를 판정하는 MOE에 목표값/한계값/단위를 강제함.
     * 3. 운용 시점 인프라 가정 & 블랙박스 상호작용 (피드백 3): 외부 환경(특히 외부 물리/조직 시스템)은
     *    운용 시점의 인프라 가정에 기반하며, SoI는 내부 구조를 숨긴 채 경계 포트 입출력으로만 상호작용함.
     */

    private import ScalarValues::*;

    // =========================================================================
    // 1. 임무 정의(Mission Triad) 및 운용효과성측정지표(MOE) 스캐폴드
    //    (교수님 피드백 1, 2 & 가이드 2장 반영)
    // =========================================================================
    package MissionAndMOEScaffold {
        doc
        /* 대상시스템의 존재 목적(What/Why)과 성공 판정 기준(MOE)을 정의하는 추상 계층 */

        // MOE 평가 방향: 클수록 좋은 지표(maximize)와 작을수록 좋은 지표(minimize) 구분
        enum def EvaluationDirectionKind {
            minimize; // 하향 지표 (예: 응답 시간, 사고율) - 평가값이 한계값 이하여야 달성
            maximize; // 상향 지표 (예: 성공률, 가용성) - 평가값이 한계값 이상이어야 달성
        }

        // [교수님 피드백 2] 목표값(Target), 한계값(Threshold), 단위(Unit)를 갖춘 MOE 구조체
        attribute def MOEMetric {
            doc
            /* 운용효과성측정지표(Measure of Effectiveness) 표준 템플릿.
             * 수치 자체의 엄밀성보다 '목표값/한계값/단위'의 정형화된 구조 정의가 핵심임.
             */
            attribute metricId : String; // 지표 식별자 (예: "MOE-01")
            attribute metricName : String; // 지표 명칭
            attribute unit : String; // 측정 단위 (예: "%", "분", "건/년", "m")
            attribute direction : EvaluationDirectionKind default EvaluationDirectionKind::maximize; // 평가 방향 (기본: 상향)
            attribute thresholdValue : Real; // 한계값 (Threshold: 임무 성공 최소 허용 기준)
            attribute targetValue : Real; // 목표값 (Objective/Target: 지향하는 이상적 기준)
            attribute evaluationCondition : String; // 측정 조건 (예: "ODD 전 구간 운용 기준")
            attribute evaluatedValue : Real [0..1]; // 시나리오/시뮬레이션 수행 후 평가값
        }

        // MOE 달성 여부 판정 제약식 (direction에 따라 부등호 방향이 결정됨)
        constraint def MOEAchievementConstraint {
            in moe : MOEMetric;
            if moe.direction == EvaluationDirectionKind::maximize ?
                moe.evaluatedValue >= moe.thresholdValue
            else
                moe.evaluatedValue <= moe.thresholdValue
        }

        // [교수님 피드백 1, 2 & 가이드 2장] Mission Triad (What / Why / Validation) 추상 요구사항
        abstract requirement def BaseMissionObjective {
            doc
            /* 모든 도메인의 최상위 임무/사업 목표 요구사항이 상속받아야 할 스캐폴드.
             * 하위 도메인에서 ':>>' (redefines)를 통해 구체적인 텍스트와 MOE를 채워 넣음.
             */
            attribute whatCoreMission : String; // [WHAT] 핵심 임무 선언 (누구를 위해 무엇을 완수하는가)
            attribute whyBusinessObjective : String; // [WHY] 존재 이유 및 사업 목표 (Pain Point 극복 및 창출 효용)
            attribute moeMetrics : MOEMetric [1..*]; // [VALIDATION] 정량적 운용효과성측정지표(MOE) 목록
        }
    } // package MissionAndMOEScaffold

    // =========================================================================
    // 2. 운용 시점(Timeframe) 인프라 가정 및 운용 장소(ODD) 그룹핑 스캐폴드

            attribute threatAndDisturbanceLevel : String [1..*]; // ③ 위협 및 외란 수준 (물리적·전자기적·기상 위협)

    package BlackBoxContextScaffold {

        doc
        /* [원칙 1: Solution-Free & Gate 2: Black-Box]
         * 대상시스템(SoI) 내부의 HW/SW 컴포넌트, 제어기, 내부 버스, 알고리즘을 일절 정의하지 않고,

        // ---- 3.1 경계면 입출력 교환 아이템 (Exchange Items) 추상 계층 ----
        abstract item def BaseBoundaryExchangeItem {

        item def OrganizationalCommandItem :> BaseBoundaryExchangeItem {

        item def OrganizationalReportItem :> BaseBoundaryExchangeItem {
            doc
            /* SoI -> 외부 조직: 운용 상태 보고, 임무 결과 로그, 긴급 구난(eCall 등) 요청 */

            /* SoI -> 외부 물리 인프라: 상태 브로드캐스트 패킷, 인프라 요청 신호, 물리적 작업 출력 등 */
        }

        // (3) 외부 자연 환경 → SoI 외란 아이템 [가이드 5장: 외란 아이템화 규칙]
        item def EnvironmentalDisturbanceItem :> BaseBoundaryExchangeItem {
            doc
            /* 자연 환경과의 상호작용을 문학적 묘사가 아닌 정형 데이터/물리 타입으로 모델링하기 위한 아이템.
             * 외부 자연 환경의 출력 타입과 SoI의 입력 타입 간 대칭성(Type Symmetry)을 보장함.
             */
            attribute disturbanceCategory : String; // 예: "노면 마찰 저하", "폭우 센서 감쇠", "태양풍 방사선"
            attribute parameterName : String; // 예: "마찰계수(mu)", "감쇠율", "열하중"

        }

            in item infraIn : PhysicalInfraInputItem [0..*];
            out item infraOut : PhysicalInfraOutputItem [0..*];
        }

        port def SoI_NaturalEnvBoundaryPort {
            in item disturbanceIn : EnvironmentalDisturbanceItem [0..*];
        }

        abstract part def BaseExternalOrganization {
            doc
            /* ① 외부 조직 (Org): 관제, 법규/인증, 소방/구난 등 인간·사회적 조직 엔티티 */
            attribute orgRole : String;
            // '~' (Conjugation)를 통해 SoI_OrganizationBoundaryPort와 입출력 방향이 정확히 대칭됨
            port orgPort : ~SoI_OrganizationBoundaryPort;
        }

        abstract part def BaseExternalPhysicalSystem {
            doc
            /* ② 외부 물리 시스템 (Physical): [교수님 피드백 3] 운용 시점의 인프라 가정이 실체화된
             * 외부 통신망, 기지국, 충전/보급 설비, 타 시스템 등 인공 물리 엔티티.

            attribute infraSystemRole : String;
            port physicalPort : ~SoI_PhysicalInfraBoundaryPort;
        }

        abstract part def BaseExternalNaturalEnvironment {
            doc
            /* ③ 외부 자연 환경 (Nature): 통제 불가능하며 시스템에 물리적·환경적 외란을 유입하는 자연계 */
            attribute environmentFactor : String;
            port naturePort : ~SoI_NaturalEnvBoundaryPort;
        }

        // ---- 3.4 블랙박스 대상시스템 (Target SoI: Whole-System & Black-Box) ----
        abstract part def BaseBlackBoxSoI {
            doc
            /* [Gate 1: Whole-System & Gate 2: Black-Box]
             * 독립적으로 임무를 완수하는 최상위 전체 시스템이며, 내부 구성요소(Part)를 갖지 않고
             * 오직 3대 외부 환경과 상호작용하는 경계 포트(Port)만 노출함.
             */
            attribute soiName : String;
            port orgBoundaryPort : SoI_OrganizationBoundaryPort [1..*];
            port physicalBoundaryPort : SoI_PhysicalInfraBoundaryPort [1..*];
            port natureBoundaryPort : SoI_NaturalEnvBoundaryPort [1..*];
        }

        // ---- 3.5 블랙박스 운용 컨텍스트 스캐폴드 (Gate 3, 4, 5 내장) ----
        abstract part def BaseBlackBoxOperationalContext {
            doc
            /* 운용 시점 전제 조건, 3대 ODD 장소 그룹, 블랙박스 SoI, 그리고
             * 각 2개 이상의 3대 외부 환경 액터(Gate 5)를 통합하는 컨텍스트 템플릿.
             */
            // [Gate 3] 운용 시점의 인프라·법규·외란 전제 조건 강제
            requirement timeframeReq {
                assume constraint timeframeAssumption : BaseTimeframeInfrastructureAssumption;
            }

            // [Gate 4] 3개 대표 장소 그룹(Group A, B, C) 포함
            part oddGroupA : OperationalLocationGroup {
                :>> groupKind = ODDStressGroupKind::GroupA_Nominal;
            }
            part oddGroupB : OperationalLocationGroup {
                :>> groupKind = ODDStressGroupKind::GroupB_ComplexHighStress;
            }
            part oddGroupC : OperationalLocationGroup {
                :>> groupKind = ODDStressGroupKind::GroupC_ExtremeDegraded;
            }

            // [Gate 1 & 2] 블랙박스 대상시스템(SoI)
            part targetSoI : BaseBlackBoxSoI;

            // [Gate 5: 3대 범위 완전성] 외부 조직, 외부 물리 시스템, 외부 자연 환경 각 2개 이상 식별 강제 ([2..*])
            part externalOrganizations : BaseExternalOrganization [2..*];
            part externalPhysicalSystems : BaseExternalPhysicalSystem [2..*];
            part externalNaturalEnvironments : BaseExternalNaturalEnvironment [2..*];
        }
    } // package BlackBoxContextScaffold

    // =========================================================================
    // 4. 블랙박스 운용 시나리오(ConOps) 및 5-Tuple 명세 스캐폴드
    //    (교수님 피드백 1, 3 & 가이드 6장, Gate 6 반영)
    // =========================================================================
    package BlackBoxConOpsScaffold {
        private import MissionAndMOEScaffold::*;
        private import TimeframeAndODDScaffold::*;
        private import BlackBoxContextScaffold::*;

        doc
        /* [교수님 피드백 1 & 3]
         * 블랙박스 대상시스템이 외부 환경(조직/물리/자연)과 상호작용하며
         * 공칭 운용 -> 외란 적응 -> 비상 폴백(MRC)을 거쳐 존재 목적(임무/MOE)을 달성하는 동적 시나리오 스캐폴드.
         */

        // 대상시스템의 임무 상태 전이(Mission State Transition) 표준 상태 정의
        enum def MissionStateKind {
            StandbyAndInitialized; // 임무 대기 및 초기 인가 상태
            NominalMissionExecuting; // 공칭(Group A) 및 복합(Group B) 환경 정상 임무 수행 상태
            DisturbanceAdapting; // 자연 외란 유입에 적응하여 성능을 유지/조정하는 상태
            MinimalRiskCondition_MRC; // 극한 환경(Group C) 또는 한계 초과 시 최소위험상태(MRC) 안전 폴백 상태
            MissionObjectiveAchieved; // 최종 임무 완수 및 사업 목표(MOE) 달성 상태
        }

        // [가이드 6장] 표준 5-Tuple 시나리오 스텝 추상 액션 정의
        abstract action def BaseScenarioStep_5Tuple {
            doc
            /* 모든 시나리오 스텝은 아래 5-Tuple 요소를 반드시 포함해야 함:
             * 1. [Step N | 장소 그룹] : stepNumber, activeODDGroup
             * 2. [상호작용 액터 (분류)] : interactingOrg / interactingPhysical / interactingNature (Gate 6 추적성)
             * 3. [교환 아이템 (방향)] : inboundItem (Actor -> SoI), outboundItem (SoI -> Actor)
             * 4. [SoI 블랙박스 반응] : soiBlackBoxReaction (내부 구현 배제, 시스템 레벨 의사결정)
             * 5. [임무 상태 전이] : transitionedState (달성된 시스템 임무 상태)
             */
            attribute stepNumber : Integer;
            attribute activeODDGroup : ODDStressGroupKind; // 1. 장소 그룹 (A/B/C)

            // 2. 상호작용 외부 액터 참조 (정의된 모든 액터가 최소 1회 이상 참조되어야 함 - Gate 6)
            ref interactingOrg : BaseExternalOrganization [0..*];
            ref interactingPhysical : BaseExternalPhysicalSystem [0..*];
            ref interactingNature : BaseExternalNaturalEnvironment [0..*];

            // 3. 경계면 교환 아이템 (방향성 명시)
            in item inboundToSoI : BaseBoundaryExchangeItem [0..*]; // [외부 환경 -> SoI] 입력/외란
            out item outboundFromSoI : BaseBoundaryExchangeItem [0..*]; // [SoI -> 외부 환경] 출력/보고

            // 4. SoI 블랙박스 반응 (Solution-Free: 내부 제어기/SW 언급 없이 경계면 반응 기술)
            attribute soiBlackBoxReaction : String;

            // 5. 임무 상태 전이 결과
            attribute transitionedState : MissionStateKind;
        }

        // [가이드 6장 & 피드백 1] 전 과정 블랙박스 운용 시나리오 (ConOps) 추상 액션
        abstract action def BaseBlackBoxConOpsScenario {
            doc
            /* 해피 패스(공칭)뿐 아니라 복합 환경 상호작용, 자연 외란 적응, 비상 폴백(MRC),
             * 그리고 최종 임무/MOE 달성 확인 국면을 시간 순서(Time-ordered sequence)로 통합한 스캐폴드.
             */
            // 국면 1: 공칭 환경(Group A) 임무 개시 및 외부 조직/인프라 상호작용
            action step1_NominalInitiation : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupA_Nominal;
                :>> transitionedState = MissionStateKind::NominalMissionExecuting;
            }

            // 국면 2: 복합/고부하 환경(Group B) 외부 물리 인프라 상호작용
            action step2_ComplexInfraInteraction : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupB_ComplexHighStress;
                :>> transitionedState = MissionStateKind::NominalMissionExecuting;
            }

            // 국면 3: 외부 자연 환경 외란(Disturbance) 유입 및 블랙박스 적응 제어
            action step3_NaturalDisturbanceAdaptation : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupB_ComplexHighStress;
                :>> transitionedState = MissionStateKind::DisturbanceAdapting;
            }

            // 국면 4: 극한/기능저하 환경(Group C) 진입 시 최소위험상태(MRC) 비상 폴백 및 조직 보고
            action step4_ExtremeDegradedMRCFallback : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupC_ExtremeDegraded;
                :>> transitionedState = MissionStateKind::MinimalRiskCondition_MRC;
            }

            // 국면 5: 임무 완수 보고 및 존재 목적(임무/사업 목표 및 MOE) 달성 완료
            action step5_MissionCompletionAndMOEReport : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupA_Nominal;
                :>> transitionedState = MissionStateKind::MissionObjectiveAchieved;
            }

            // 시계열 제어 흐름 (하위 도메인에서 세부 스텝 추가/분기 가능)
            first step1_NominalInitiation;
            then step2_ComplexInfraInteraction;
            then step3_NaturalDisturbanceAdaptation;
            then step4_ExtremeDegradedMRCFallback;
            then step5_MissionCompletionAndMOEReport;
        }
    } // package BlackBoxConOpsScaffold

    // =========================================================================
    // 5. [사용 가이드] 각 도메인 워킹그룹용 상속(Specialization) 스켈레톤 템플릿
    // =========================================================================
    package DomainWG_Template_Skeleton {
        doc
        /* 각 워킹그룹은 본 패키지를 복사하여 패키지명과 '<...>' 부분만 도메인에 맞게 채워 넣으면
         * 교수님 피드백 3가지와 6대 품질 게이트(Gate 1~6)를 자동으로 준수할 수 있음.
         */
        private import MissionAndMOEScaffold::*;

        private import BlackBoxConOpsScaffold::*;

        // [Step A] 도메인 임무/사업목표 및 MOE 정의 (BaseMissionObjective 상속)
        requirement def DomainMissionObjective :> BaseMissionObjective {
            :>> whatCoreMission = "<누구를 위해 무엇을 완수하는지 단일 선언문 작성>";
            :>> whyBusinessObjective =
                "<극복할 기존 한계(Pain Point) 및 창출할 사업적/사회적 가치 작성>";

            attribute primaryMOE : MOEMetric :> moeMetrics {
                :>> metricId = "MOE-01";
                :>> metricName = "<운용효과성측정지표 명칭>";
                :>> unit = "<단위: %, 분, 건/년 등>";
                :>> direction = EvaluationDirectionKind::maximize; // 하향 지표(작을수록 좋음)이면 minimize로 변경
                :>> thresholdValue = 0.0; // <한계값 입력>
                :>> targetValue = 0.0; // <목표값 입력>
                :>> evaluationCondition = "<ODD 내 측정 조건 입력>";
            }
        }

        // [Step B] 운용 시점의 외부 환경/인프라 4대 가정 정의 (BaseTimeframeInfrastructureAssumption 상속)
        constraint def DomainTimeframeAssumption :> BaseTimeframeInfrastructureAssumption {
            :>> targetTimeframe = "<운용 목표 연도 및 시점 정의>";
            :>> externalInfraMaturity =
                "<운용 시점에 구축되어 있을 외부 통신/전력/설비 인프라 가정>";
            :>> regulatoryEnvironment = "<운용 시점의 인허가/법규/책임 기준 가정>";
            :>> threatAndDisturbanceLevel = "<운용 시점의 주요 물리적/환경적/사이버 위협 가정>";
            :>> lifecycleEvolutionTrait = "<배포 후 수명주기 동안의 진화/유지보수 가정>";
        }

        // [Step C] 블랙박스 SoI 및 블랙박스 ConOps 시나리오 정의
        part def DomainBlackBoxSoI :> BaseBlackBoxSoI {
            :>> soiName =
                "<하위 부품이 아닌 독립 임무 완결형 최상위 대상시스템(Whole-System) 명칭>";
        }

        action def DomainConOpsScenario :> BaseBlackBoxConOpsScenario {
            // step1 ~ step5의 5-Tuple 속성을 도메인 상호작용에 맞게 재정의(:>>)
        }

        // [Step D] 도메인 운용 컨텍스트 통합 및 임무 만족(satisfy) 연결
        part def DomainOperationalContext :> BaseBlackBoxOperationalContext {
            requirement domainMission : DomainMissionObjective;

            :>> targetSoI : DomainBlackBoxSoI;
            perform action conOpsScenario : DomainConOpsScenario;

            // externalOrganizations, externalPhysicalSystems, externalNaturalEnvironments를
            // 각 2개 이상 구체화하고 connect 문으로 targetSoI의 경계 포트와 연결

            // [교수님 피드백 1] 대상시스템(SoI) 및 운용시나리오(ConOps)가 도메인 임무 목표를 만족함
            satisfy domainMission by targetSoI;
            satisfy domainMission by conOpsScenario;
        }
    } // package DomainWG_Template_Skeleton
} // package LV1_BlackBox_ConOps_Scaffold

##3_ASELCM_LV1_Target_Reference_Models/0_Common_Foundation/2_Logical_WhiteBox_Scaffold.sysml##
package LV1_Logical_WhiteBox_Scaffold {
    doc
    /* [InKC LV1 공통 스캐폴드 패키지 - Stage 3: 논리 화이트박스]
     * Stage 2(블랙박스/ConOps)에서 정의한 대상시스템(SoI)을 상속받아
     * 내부 논리 서브시스템과 논리 기능으로 분해(Decomposition)하고,
     * 논리 기능을 논리 서브시스템에 할당(allocate)하는 추상 메타모델.
     *
     * [설계 원칙]
     * 1. 블랙박스 경계(포트)는 BaseBlackBoxSoI로부터 그대로 상속하여 외부 인터페이스를 보존함.
     * 2. 논리 계층은 Solution-Free: 구체적 HW/SW 구현 대신 역할 중심의 논리 요소만 정의함.
     * 3. 모든 논리 기능은 최소 1개 논리 서브시스템에 할당되어야 함 (기능-구조 추적성).
     */

    private import ScalarValues::*;
    private import LV1_BlackBox_ConOps_Scaffold::*;
    private import LV1_BlackBox_ConOps_Scaffold::BlackBoxContextScaffold::*;

    // ---- 1. 논리 기능 (Logical Function) ----
    abstract action def BaseLogicalFunctionAction {
        doc
        /* 대상시스템 내부에서 수행되는 논리 기능의 추상 정의.
         * 블랙박스 ConOps 시나리오 스텝의 SoI 반응을 내부 기능 수준으로 구체화함.
         */
        attribute functionName : String;
        attribute functionDescription : String [0..1];
        in item functionInput : BaseBoundaryExchangeItem [0..*];
        out item functionOutput : BaseBoundaryExchangeItem [0..*];
    }

    // ---- 2. 논리 서브시스템 (Logical Subsystem) ----
    abstract part def BaseLogicalSubsystem {
        doc
        /* 대상시스템을 구성하는 논리적 서브시스템(역할 단위)의 추상 정의.
         * 물리적 구현 기술을 가정하지 않으며, 물리 요소로의 할당은 Stage 4에서 수행함.
         */
        attribute subsystemName : String;
        attribute logicalResponsibility : String [0..1];
    }

    // ---- 3. 화이트박스 논리 대상시스템 ----
    abstract part def BaseWhiteBoxLogicalSoI :> BaseBlackBoxSoI {
        doc
        /* 블랙박스 SoI의 경계 포트를 유지한 채 내부를 논리 서브시스템과 논리 기능으로 분해한 화이트박스 SoI.
         * 하위 도메인은 logicalSubsystems / logicalFunctions를 구체 타입으로 재정의(:>>)하고
         * 기능-서브시스템 할당을 구체화함.
         */
        part logicalSubsystems : BaseLogicalSubsystem [1..*];
        perform action logicalFunctions : BaseLogicalFunctionAction [1..*];

        // 논리 기능 -> 논리 서브시스템 할당 (도메인에서 개별 기능/서브시스템 쌍으로 세분화)
        allocate logicalFunctions to logicalSubsystems;
    }
} // package LV1_Logical_WhiteBox_Scaffold

        doc
        /* 논리 화이트박스 SoI를 물리 컴포넌트로 실체화한 SoI.
         * 하위 도메인은 physicalComponents를 구체 타입으로 재정의하고 할당을 세분화함.
         */
        part physicalComponents : BasePhysicalComponent [1..*];

        // 논리 서브시스템 -> 물리 컴포넌트 할당 (도메인에서 개별 쌍으로 세분화)
        allocate logicalSubsystems to physicalComponents;
    }

    // ---- 3. 시스템 검증 케이스 (System Verification Case) ----
    abstract verification def BaseSystemVerificationCase {
        doc
        /* 물리 SoI가 임무 요구사항을 달성하는지 MOE 기준으로 검증하는 표준 검증 케이스.
         * 하위 도메인은 검증 대상 MOE(moeUnderTest)와 평가 방법을 구체화함.
         */
        subject systemUnderTest : BasePhysicalSoI;

        // 검증 대상 MOE 및 판정 제약 (방향성은 MOEMetric.direction에 따라 결정됨)
        in attribute moeUnderTest : MOEMetric;
        constraint moeCheck : MOEAchievementConstraint {
            in :>> moe = moeUnderTest;
        }

        objective moeVerificationObjective {
            doc
            /* 임무 요구사항의 MOE가 한계값 기준을 만족함을 검증 */
            verify requirement missionRequirement : BaseMissionObjective;
        }
    }
} // package LV1_Physical_and_VnV_Scaffold
