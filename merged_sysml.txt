package INCOSE_Korea_Chapter_Governance {
    doc
    /* ASELCM System 3 (Environment & Governance System)
     * 통제 대상:
     *   - System 2 (LV2: INCOSE Korea Chapter 조직 및 과제 운영 모델)
     *   - System 1 (LV1: 워킹그룹별 도메인 참조모델 - 3_ASELCM_LV1_Target_Reference_Models)
     * 목적: 챕터 운영 모델과 도메인 참조모델의 일관성, 확장성, 표준 준수성 및 학술적 엄밀성을 통제하는 거버넌스 프레임워크
     */

    private import ScalarValues::*;
    private import SystemContext::*;
    private import LV1_BlackBox_ConOps_Scaffold::*;

    // ==========================================
    // 1. Modeling Conventions (모델링 규약, 스캐폴드 상속 및 형상 관리)
    // ==========================================
    package ModelingConventions {
        doc
        /* System-as-Code 접근 방식에 따른 SysML v2 모델 작성 표준 및 버전 관리 체계.
         * Git/GitHub 기반의 브랜치 전략, 명명 규칙, 그리고 LV1 공통 스캐폴드 상속 규칙을 정의하여
         * 다수 회원의 협업 충돌을 방지하고 도메인 간 메타모델 일관성을 유지함.
         */

        constraint def NamingConventionRule {
            doc
            /* 1) 모든 Definition(part def, action def, item def 등)은 PascalCase 준수
             * 2) 모든 Usage 및 Attribute는 camelCase 준수
             * 3) 주석(doc)으로만 정보를 남기지 않고 정형 속성(attribute)과 참조(ref)로 모델링
             */
            attribute isPascalCaseDef : Boolean default true;
            attribute isCamelCaseUsage : Boolean default true;
        }

        constraint def VersionControlPolicy {
            doc
            /* 메인 브랜치(main) 및 KnowledgeRepository 병합 전:
             * 1) 워킹그룹 리드(WG Lead)의 1차 모델 검토
             * 2) 학술자문단(Academic Advisors: 교수진)의 방법론/피드백 검증
             * 3) 보안 및 형상 관리 Pull Request 승인 필수
             */
            attribute isWGLeadReviewed : Boolean;
            attribute isAcademicAdvisorApproved : Boolean;
            attribute isSecretaryMerged : Boolean;
            isWGLeadReviewed and isAcademicAdvisorApproved and isSecretaryMerged
        }

        constraint def ScaffoldInheritanceRule {
            doc
            /* [LV1 참조모델 아키텍처 일관성 규칙]
             * 3_ASELCM_LV1_Target_Reference_Models 내 모든 워킹그룹 모델은
             * 'LV1_BlackBox_ConOps_Scaffold'의 추상 정의(BaseMissionObjective, BaseBlackBoxSoI,
             * BaseBlackBoxOperationalContext, BaseBlackBoxConOpsScenario)를 상속(:>)하여 구현해야 함.
             */
            attribute inheritsCommonScaffold : Boolean;
            attribute maintainsPortTypeSymmetry : Boolean; // 외부 액터 포트(~)와 SoI 포트 간 대칭성 유지 여부
            inheritsCommonScaffold and maintainsPortTypeSymmetry
        }
    } // package ModelingConventions

    // =========================================================================
    // 2. Policy & Directives (글로벌 정책, SE 국제 표준 및 보안 규정)
    // =========================================================================
    package PolicyAndDirectives {
        doc
        /* INCOSE Global (Board of Directors), Asia-Oceania Sector 상위 지침 및
         * 국제 SE 표준(ISO 15288, GtWR v4), 기업 정보보안 규정을 강제하는 정책 패키지.
         */

        requirement def SectorComplianceRule {
            doc
            /* 아시아-오세아니아 섹터(Sector III) 정기 활동 보고 양식 및 주기 준수 규정 */
        }

        requirement def ToolLicensePolicy {
            doc
            /* 상용 MBSE 툴(Cameo 등), Syside 및 SysML v2 API 사용 시 라이선스 위반 방지 규정 */
        }

        requirement def SEStandardAlignmentPolicy {
            doc
            /* [실무 가이드 표준 근거 반영]
             * 모든 LV1 참조모델 및 LV2 요구사항은 아래 3대 국제 표준을 준거 틀로 삼아야 함:
             * 1) ISO/IEC/IEEE 15288 (시스템 생명주기 프로세스 - 문제 공간과 해결 공간의 분리)
             * 2) INCOSE GtWR v4 (Guide to Writing Requirements - What/Why/Validation 3대 축)
             * 3) OMG SysML v2 (정형 텍스트 및 그래픽 모델링 표준 문법)
             */
        }

        requirement def EnterpriseSecurityAndIPPolicy {
            doc
            /* [킥오프 합의 및 회원 보안 요구사항 반영]
             * 소속 기업의 지적 재산(IP) 및 방산/산업 민감 데이터 보호를 위해:
             * 1) 사내 실제 수치 및 프로젝트명/회사명 직접 기재 절대 금지
             * 2) 인터넷 공개 데이터 및 AI 기반 가상 치환 데이터(Anonymized Data)만 사용
             */
            attribute isInternalDataExcluded : Boolean;
            attribute isCompanyProjectAnonymized : Boolean;
        }
    } // package PolicyAndDirectives

    // =========================================================================
    // 3. Quality Assurance & Maturity Gates (품질 보증, 교수님 피드백 & 6대 게이트)
    // =========================================================================
    package QualityAssurance {
        doc
        /* System 2(챕터 운영)의 성장 단계 통제(Maturity Gate)와
         * System 1(LV1 도메인 참조모델 과제)의 무결성 검증(교수님 피드백 3대 원칙 + 실무 가이드 6대 Gate)을
         * 수행하는 품질 보증 패키지.
         */

        // ---------------------------------------------------------------------
        // 3.1 [교수님 피드백 1·2·3 반영] 학술 자문단 3대 핵심 모델링 검증 규칙
        // ---------------------------------------------------------------------
        constraint def Feedback1_MissionAchievementConOpsRule {
            doc
            /* [교수님 피드백 1]
             * 운용 개념(ConOps)은 단순 기능 나열이 아니라 대상시스템(SoI)의 존재 목적(임무 또는 사업목표)을
             * 달성하는 모습을 표현해야 하며, 시나리오 수행 결과가 임무 요구사항을 satisfy해야 함.
             */
            attribute expressesMissionAndBusinessGoal : Boolean;
            attribute hasExplicitSatisfyLinkToMission : Boolean;
            expressesMissionAndBusinessGoal and hasExplicitSatisfyLinkToMission
        }

        constraint def Feedback2_QuantitativeMOEStructureRule {
            doc
            /* [교수님 피드백 2]
             * 임무/사업 목표에 대한 서술(What/Why)과 함께, 달성 여부를 판정하는
             * 운용효과성측정지표(MOE)를 정의하고 반드시 목표값(Target), 한계값(Threshold), 단위(Unit)를
             * 지정해야 함 (수치 자체의 정확성보다 정량적 구조 완결성이 필수).
             */
            attribute hasMissionAndBusinessNarrative : Boolean;
            attribute hasMOETargetValue : Boolean;
            attribute hasMOEThresholdValue : Boolean;
            attribute hasMOEUnit : Boolean;
            hasMissionAndBusinessNarrative and
            hasMOETargetValue and
            hasMOEThresholdValue and
            hasMOEUnit
        }

        constraint def Feedback3_BlackBoxAndTimeframeInfraRule {
            doc
            /* [교수님 피드백 3]
             * 운용 시나리오 작성 시 블랙박스 개념을 철저히 유지하여 블랙박스 대상시스템이 외부 환경과
             * 상호작용하는 것에 초점을 두어야 하며, 이때 외부 환경에 대한 가정은 반드시
             * '운용 시점의 인프라에 대한 가정(Timeframe Infrastructure Assumption)'에 기반해야 함.
             */
            attribute focusesOnExternalBoundaryInteraction : Boolean;
            attribute isExternalEnvGroundedInTimeframeInfra : Boolean;
            focusesOnExternalBoundaryInteraction and isExternalEnvGroundedInTimeframeInfra
        }

        // ---------------------------------------------------------------------
        // 3.2 [실무 가이드 7장 반영] 과제 산출물 6대 자체 감사 게이트 (Verification Gates)
        // ---------------------------------------------------------------------
        constraint def Gate1_WholeSystemScopeRule {
            doc
            /* [Gate 1: Whole-System]
             * 대상시스템(SoI)이 하위 모듈/센서/단일 제어기가 아닌, 독립적으로 이해관계자에게
             * 임무를 완수할 수 있는 최상위 전체 시스템(Whole-System)으로 선정되었는가?
             */
            attribute isIndependentWholeSystem : Boolean;
            isIndependentWholeSystem
        }

        constraint def Gate2_BlackBoxSolutionFreeRule {
            doc
            /* [Gate 2: Black-Box (Solution-Free 원칙)]
             * 내부 아키텍처(특정 MCU, ECU, SW 함수, 내부 통신 버스 규격, 알고리즘)가 철저히 배제되고
             * 경계면의 입출력(Port/Item)만 노출되었는가?
             */
            attribute hasZeroInternalComponentsExposed : Boolean;
            hasZeroInternalComponentsExposed
        }

        constraint def Gate3_TimeframeAssumptionLinkRule {
            doc
            /* [Gate 3: 시점 전제 연계]
             * 단순 연도 텍스트 기재를 넘어, 운용 시점(예: 2030~2040년)의 4대 전제 조건
             * (①외부 인프라 성숙도, ②법규/제도, ③위협/외란 수준, ④수명주기 진화 특성)이 반영되었는가?
             */
            attribute hasFourCategoryTimeframeAssumptions : Boolean;
            hasFourCategoryTimeframeAssumptions
        }

        constraint def Gate4_ODDSpatialGroupingRule {
            doc
            /* [Gate 4: 장소 그룹핑]
             * 무분별한 개별 지명 나열 대신, 물리적·기능적 부하(Stress)의 동질성을 기준으로
             * 그룹 A(공칭/정규), 그룹 B(복합/고부하), 그룹 C(극한/기능저하)로 추상화되었는가?
             */
            attribute hasThreeTierODDGrouping : Boolean;
            hasThreeTierODDGrouping
        }

        constraint def Gate5_ThreeExternalScopesCompletenessRule {
            doc
            /* [Gate 5: 3대 범위 완전성 및 외란 아이템화]
             * 외부 조직(Org), 외부 물리 시스템(Physical), 외부 자연 환경(Nature)이 각각 2개 이상 식별되었으며,
             * 자연 환경 외란이 문학적 묘사가 아닌 EnvironmentalDisturbanceItem으로 정형화되었는가?
             */
            attribute externalOrgCount : Integer;
            attribute externalPhysicalCount : Integer;
            attribute externalNatureCount : Integer;
            attribute isNaturalDisturbanceItemized : Boolean;

            externalOrgCount >= 2 and
            externalPhysicalCount >= 2 and
            externalNatureCount >= 2 and
            isNaturalDisturbanceItemized
        }

        constraint def Gate6_NoIsolatedActorTraceabilityRule {
            doc
            /* [Gate 6: 고립 액터 방지 (Traceability 강제)]
             * 컨텍스트에 정의된 모든 외부 액터(조직/물리/자연)와 시점·장소 그룹이
             * 최종 5-Tuple 운용 시나리오(공칭 + 외란 적응 + 비상 폴백/MRC)에 최소 1회 이상 등장하는가?
             */
            attribute hasZeroIsolatedExternalActors : Boolean;
            attribute includesNominalDisturbanceAndMRCFlows : Boolean;
            hasZeroIsolatedExternalActors and includesNominalDisturbanceAndMRCFlows
        }

        // ---------------------------------------------------------------------
        // 3.3 LV1 참조모델 과제 통합 승인 게이트 (교수님 피드백 3종 + 가이드 Gate 6종 통합)
        // ---------------------------------------------------------------------
        constraint def ReferenceModelAssignmentApprovalGate {
            doc
            /* 워킹그룹이 제출한 LV1 모델이 KnowledgeRepository에 정식 Baseline으로 승격되기 위한 통합 품질 게이트 */
            constraint fb1 : Feedback1_MissionAchievementConOpsRule;
            constraint fb2 : Feedback2_QuantitativeMOEStructureRule;
            constraint fb3 : Feedback3_BlackBoxAndTimeframeInfraRule;

            constraint gate1 : Gate1_WholeSystemScopeRule;
            constraint gate2 : Gate2_BlackBoxSolutionFreeRule;
            constraint gate3 : Gate3_TimeframeAssumptionLinkRule;
            constraint gate4 : Gate4_ODDSpatialGroupingRule;
            constraint gate5 : Gate5_ThreeExternalScopesCompletenessRule;
            constraint gate6 : Gate6_NoIsolatedActorTraceabilityRule;
        }

        // ---------------------------------------------------------------------
        // 3.4 모델 검수 및 승인 프로세스 (학술 자문단 리뷰 및 6대 Gate 검증 반영)
        // ---------------------------------------------------------------------
        action def ModelReviewProcess {
            doc
            /* 정기적인 챕터 모델 및 LV1 과제 산출물 검수·승인 워크플로우 */
            action submitModelUpdate {
                doc
                /* 워킹그룹 또는 회원이 스캐폴드 기반 모델 제출 */
            }
            action checkAutomatedConventionsAndGates {
                doc
                /* 6대 자체 감사 게이트(Gate 1~6) 및 스캐폴드 상속·포트 대칭성 1차 점검 */
            }
            action reviewByAcademicAdvisors {
                doc
                /* 교수진 자문단(이중윤·서은석 교수)의 피드백 1·2·3 충족 여부 심사 */
            }
            action reviewByChapterLeadership {
                doc
                /* 챕터 대표 및 총무의 보안 규정(사내 데이터 배제) 확인 및 최종 승인 */
            }
            action mergeAndDeploy {
                doc
                /* GitHub 메인 브랜치 및 Google Workspace 단일 진실 공급원(SSOT) 배포 */
            }

            first submitModelUpdate;
            then checkAutomatedConventionsAndGates;
            then reviewByAcademicAdvisors;
            then reviewByChapterLeadership;
            then mergeAndDeploy;
        }

        // ---------------------------------------------------------------------
        // 3.5 System 2 (챕터 조직) 성장 단계 전환 조건 (Maturity Gate)
        // ---------------------------------------------------------------------
        constraint def PhaseTransitionGateCondition {
            doc
            /* 조직 성장 조건 (Maturity Gate):
             * LV1 참조모델 승인 누적 건수 및 세미나 발표 건수 등 정량적 지표를 기반으로
             * System 2(KoreaChapter)의 Lifecycle Phase 전환 가능 여부를 판정함.
             */
            attribute approvedReferenceModelCount : Integer;
            attribute accumulatedSeminarArtifactCount : Integer;
            attribute activeMemberAttendanceCount : Integer;
        }
    } // package QualityAssurance

    // =========================================================================
    // 4. Tooling Ecosystem (도구 사슬 및 IT 환경)
    // =========================================================================
    package ToolingEcosystem {
        doc
        /* 챕터 모델이 작성, 검증, 배포되고 시각화되는 외부 IT 인프라 환경. */

        part def ModelRepositoryServer {
            doc
            /* GitHub 및 Google Workspace 기반 SysML v2 모델 호스팅 및 버전 관리 서버 */
        }

        part def VisualizationEngine {
            doc
            /* Syside / Tom Sawyer 등 텍스트 모델을 다이어그램 및 테이블/매트릭스로 렌더링하는 환경 */
        }

        part def SimulationAndCodeGenPipeline {
            doc
            /* [ChapterObjective 3번 연계]
             * SysML v2 모델로부터 MILS/SILS/HILS 시뮬레이션 및 Simulink 오토코드 제너레이션까지
             * End-to-End 추적성을 검증하는 도구 사슬.
             */
        }
    } // package ToolingEcosystem

    // =========================================================================
    // 5. Governance System (System 3 실체 및 통제 인터페이스)
    // =========================================================================
    // 거버넌스 포트 교환 아이템 정의
    item def GovernancePolicyDirectiveItem {
        doc
        /* System 3 -> System 2 & System 1: 모델링 규약, 스캐폴드 상속 지침, 보안 규정, 6대 Gate 기준 */
        attribute policyVersion : String;
        attribute targetAssignmentStage : String;
    }

    item def ChapterComplianceAndMetricsItem {
        doc
        /* System 2 & System 1 -> System 3: 과제 게이트 통과 결과, 승인 모델 수, 운영 지표 보고 */
        attribute submittedModelName : String;
        attribute isAssignmentGatePassed : Boolean;
        attribute currentApprovedModelsCount : Integer;
    }

    part def GovernanceSystem {
        doc
        /* System 2(KoreaChapter)와 System 1(LV1 참조모델)을 감싸고 통제하는 System 3의 최상위 컨텍스트.
         * 거버넌스 규칙과 품질 게이트(교수님 피드백 + 6대 Gate)를 하달하고(out),
         * 규정 준수 상태 및 과제 검증 결과를 수집(in)하여 승인 및 조직 성장(Maturity Gate)을 통제함.
         */

        // 1) 모델링 규약 및 스캐폴드 상속 원칙
        part namingConventions : ModelingConventions::NamingConventionRule [1..*];
        part vcsPolicy : ModelingConventions::VersionControlPolicy;
        part scaffoldRule : ModelingConventions::ScaffoldInheritanceRule;

        // 2) 글로벌 정책, 국제 SE 표준 및 기업 보안 규정
        part sectorDirectives : PolicyAndDirectives::SectorComplianceRule [1..*];
        part standardPolicy : PolicyAndDirectives::SEStandardAlignmentPolicy;
        part securityPolicy : PolicyAndDirectives::EnterpriseSecurityAndIPPolicy;

        // 3) 품질 보증 프로세스 및 게이트 (교수님 피드백 3종 + 가이드 6대 Gate + 챕터 성장 Gate)
        part qaProcess : QualityAssurance::ModelReviewProcess;
        part lv1AssignmentGate : QualityAssurance::ReferenceModelAssignmentApprovalGate;
        part chapterMaturityGate : QualityAssurance::PhaseTransitionGateCondition;

        // 4) 도구 생태계 인프라
        part repositoryServer : ToolingEcosystem::ModelRepositoryServer;
        part visualizer : ToolingEcosystem::VisualizationEngine;
        part simPipeline : ToolingEcosystem::SimulationAndCodeGenPipeline [0..1];

        // System 2(KoreaChapter) 및 System 1(LV1 Models)을 통제하기 위한 인터페이스 포트
        port governanceControlPort {
            doc
            /* 거버넌스 정책(스캐폴드 규약, 교수님 피드백 기준, 6대 검증 게이트, 보안 규칙)을 하달 */
            out item policyDirectives : GovernancePolicyDirectiveItem;
        }

        port complianceMonitoringPort {
            doc
            /* LV1 과제 게이트 통과 여부 및 LV2 운영 지표를 수집하여 Baseline 승인 및 Phase 전환 심사 */
            in item chapterOperationalMetrics : ChapterComplianceAndMetricsItem;
        }
    }
} // package INCOSE_Korea_Chapter_Governance

##2_ASELCM LV2_Chapter_System/1_View and Viewpoint/1. Viewpoint.sysml##
package Viewpoints {
    private import StakeholdersConcerns::*;

    viewpoint 'administrative operations viewpoint' {
        frame concern : OperationalFlowConcern;
        require constraint {

package ChapterBehavior {
    doc
    /* Chapter Behavior — 통합된 시스템 경계 및 인터페이스 모델을 바탕으로

    // ==========================================
    // 1. ConOps (운용 시나리오 및 프로세스)
    // ==========================================
    package ConOps {
        doc
        /* 온라인 지식교환 방식, Do-Tank 참조모델 개발 프로세스 및 상위 시스템 보고 프로세스의 동적 행위 */

        enum def ExchangeModeKind {
            seminarStyle; // 세미나/특강형
            caseStudyStyle; // 케이스 스터디형
            handsOnLabStyle; // 프로젝트 실습형
            literatureReadingStyle; // 문헌 리딩형
        }

        action def KnowledgeExchangeSession {
            doc
            /* 내부 비동기/실시간 지식 교환 프로세스 (Internal Flow) */

            in attribute mode : ExchangeModeKind;
            in item referenceResource : HandbookResourceItem [0..1]; // TechOps 등 외부 문헌 리소스 활용

            out item producedArtifact : ChapterArtifact; // 논의 후 생성된 지식 자산

            action present;
            action discuss;
            action archiveToKnowledgeStore {
                doc
                /* 생성된 산출물(producedArtifact)을 KnowledgeRepository 단일 진실 공급원에 저장 */
            }

            // 제어 흐름 (Control Flow)
            first present;
            then discuss;
            then archiveToKnowledgeStore;
        }

        // [신규] Do-Tank 핵심 운영 프로세스: SE 과제 수행 및 LV1 참조모델 개발·검증 워크플로우
        action def ReferenceModelDevelopmentProcess {
            doc
            /* [LV2 Do-Tank 참조모델 개발 및 품질 게이트 프로세스]
             * 1) 과제 가이드(PDF) 및 LV1 공통 스캐폴드(LV1_BlackBox_ConOps_Scaffold) 배포
             * 2) 워킹그룹별 도메인 모델링 수행 (블랙박스 SoI, 임무/MOE, 시점 인프라 가정, 5-Tuple 시나리오)
             * 3) 학술자문단(교수진) 피드백 3대 원칙 및 가이드 6대 자체 감사 게이트(Gate 1~6) 심사
             * 4) 기업 보안 규정(사내 데이터 배제 및 가상값 치환) 준수 확인
             * 5) KnowledgeRepository 및 GitHub 메인 브랜치 Baseline 정식 등록
             */
            in item assignmentGuide : ChapterArtifact;
            out item approvedBaselineModel : ReferenceModelArtifact;

            action issueAssignmentAndScaffold {
                doc
                /* 과제 목표, 실무 가이드 및 공통 스캐폴드 코드 배포 */
            }
            action developDomainModelInWG {
                doc
                /* 워킹그룹이 스캐폴드를 상속(:>)하여 도메인 모델 작성 */
            }
            action reviewByAdvisorsAndVerifyGates {
                doc
                /* 교수님 피드백 1(존재목적 달성 표현), 피드백 2(MOE 목표값/한계값/단위),
                 * 피드백 3(운용시점 인프라 가정 및 블랙박스 상호작용) 및 Gate 1~6 충족 여부 검증
                 */
            }
            action verifySecurityAndAnonymization {
                doc
                /* 회사명/프로젝트명 비언급 및 민감 수치 가상값 치환 여부 점검 */
            }
            action baselineToKnowledgeRepository {
                doc
                /* 검증 통과 모델을 KnowledgeRepository::approvedReferenceModels에 정식 등록 */
            }

            first issueAssignmentAndScaffold;
            then developDomainModelInWG;
            then reviewByAdvisorsAndVerifyGates;
            then verifySecurityAndAnonymization;
            then baselineToKnowledgeRepository;
        }

        action def SectorReporting {
            doc
            /* 상위 시스템(INCOSE Asia-Oceania Sector)으로의 활동 결과 보고 프로세스 (External Flow).
             * 챕터 총무 및 운영진이 내부 지식 저장소의 산출물들을 취합하여 공식 리포트 형태로 변환 및 전송함.
             */

            in item internalData : ChapterArtifact [1..*]; // 지식 저장소에 누적된 내부 산출물
            out item activityReport : SectorActivityReportItem; // Sector 전송용 포맷

            first compileReport;
            then sendToSector;
        }
    } // package ConOps

    // ==========================================
    // 2. Lifecycle (수명주기 상태머신)
    // ==========================================
    package Lifecycle {
        doc
        /* 챕터 성장 단계별 상태머신.
         * LV3 거버넌스의 PhaseTransitionGateCondition(승인된 참조모델 3건 이상, 발표자료 10건 이상)과 연동됨.
         */

        state def ChapterLifecycle {
            attribute approvedModelsCount : Integer default 0;
            attribute seminarArtifactsCount : Integer default 0;

            entry;
            then onlineStudyPhase;

            state onlineStudyPhase {
                doc
                /* Phase 1: 온라인 스터디 및 도메인별 LV1 참조모델 초기 개발 단계 (현재)
                 * - 활성 Context 3종만 연결된 상태
                 * - 워킹그룹별 블랙박스 정의 및 ConOps 모델링 과제 수행 중
                 */
            }

            state sponsorshipAndOfflineSeminarPhase {
                doc
                /* Phase 2: 기업 스폰서 확보 및 오프라인 세미나 정례화 단계
                 * - 교육/실무 참고용 LV1 참조모델 3건 이상 확보 및 발표자료 10건 이상 누적 시 진입
                 * - Sponsor, PeerSociety(KOSSE/KCOSE) 연결 활성화
                 */
            }

            state officialWorkingGroupPhase {
                doc
                /* Phase 3: INCOSE 본부 공식 워킹그룹(WG) 발족 및 자체 아키텍트 인증 프로그램 운영 단계 */
            }

            transition phase1To2
                first onlineStudyPhase
                if approvedModelsCount >= 3 and seminarArtifactsCount >= 10
                then sponsorshipAndOfflineSeminarPhase {
                doc
                /* Phase 2 전환(Transition) 조건 (LV3 PhaseTransitionGateCondition 반영):
                 * 승인된 도메인 참조모델(ReferenceModelArtifact) 3건 이상 & 누적 세미나/발표 자료 10건 이상 달성
                 */
            }

            transition phase2To3
                first sponsorshipAndOfflineSeminarPhase then officialWorkingGroupPhase {
                doc
                /* Phase 3 전환(Transition) 조건:
                 * 정기 오프라인 세미나 4회 이상 개최, 기업 스폰서 1곳 이상 확보 및 CSEP 보유 인원 10명 이상 달성
                 */
            }
        }
    } // package Lifecycle
} // package ChapterBehavior

    // 1. 5대 핵심 성과 지표 홀더 (Core 5 MoEs Holder)
    // =========================================================================
    part def ChapterMoEsHolder {
        doc
        /* KoreaChapter(SoI)가 상속받는 5대 핵심 정량 지표(MoEs) */

        attribute accumulatedPresentationsCount : Integer default 1; // MoE 1: 누적 발표/사례 자료 수 (목표 >= 10)
        attribute completedReferenceModelsCount : Integer default 0; // MoE 2: 완성된 도메인 참조모델 수 (목표 >= 3)
        attribute developedAISkillsCount : Integer default 0; // MoE 3: 단계별 AI Skills 수 (Phase2 >= 3, Phase3 >= 5)
        attribute regularActiveMembersCount : Integer default 12; // MoE 4: 고정 활동 인원 수 (Phase2 > 15, Phase3 >= 30)
        attribute sepCertifiedMembersCount : Integer default 1; // MoE 5: SEP 자격 보유자 수 (Phase2 >= 3, Phase3 >= 10)
    }

    // =========================================================================
    // 2. Phase 1 -> Phase 2 전환 게이트 제약 (온라인 스터디 -> 오프라인 세미나 정례화)
    // =========================================================================
    constraint def Phase1ToPhase2GateConstraint :> PhaseTransitionGateCondition {
        doc
        /* [Phase 1 -> Phase 2 전환 조건 (기업 스폰서 조건 제외)]
         * 1) 누적 발표/사례 자료 10건 이상 OR 고정 활동 인원 15명 초과
         * 2) 도메인별 참조모델 3건 이상 완성
         * 3) 단계별 AI Architecting Skills 3개 이상 구축
         * 4) INCOSE SEP 자격 보유 인원 3명 이상 확보
         */
        in presentations : Integer;
        in refModels : Integer;
        in aiSkills : Integer;
        in activeMembers : Integer;
        in sepMembers : Integer;

        (presentations >= 10 or activeMembers > 15) and
        refModels >= 3 and
        aiSkills >= 3 and
        sepMembers >= 3
    }

    // =========================================================================
    // 3. Phase 2 -> Phase 3 전환 게이트 제약 (세미나 정례화 -> 공식 WG 및 학회 Joint WG)
    // =========================================================================

         * 1) 고정 활동 인원 30명 이상 확대 (국내 학회 산하 Joint WG 연계 기준)

         * 3) MagicGrid 전 단계(5단계) AI Architecting Skills 5개 이상 완비
         * 4) 도메인별 참조모델 3건 이상 유지 및 고도화
         */
        in activeMembers : Integer;
        in sepMembers : Integer;
        in aiSkills : Integer;
        in refModels : Integer;

        activeMembers >= 30 and sepMembers >= 10 and aiSkills >= 5 and refModels >= 3
    }

package LV1_BlackBox_ConOps_Scaffold {
    doc
    /* [InKC LV1 공통 스캐폴드 패키지]
     * 각 도메인 워킹그룹(자동차, 우주, 스마트팩토리, 반도체장비 등)이 상속(:>)하여
     * 대상시스템(SoI)의 블랙박스 정의와 운용개념(ConOps)을 일관되게 구축하기 위한 추상 메타모델.
     *
     * [핵심 설계 원칙]
     * 1. 존재 목적 달성 표현 (피드백 1): ConOps 시나리오가 임무/사업 목표(MissionRequirement)를 satisfy함.
     * 2. 정량적 MOE 구조화 (피드백 2): 임무 달성 여부를 판정하는 MOE에 목표값/한계값/단위를 강제함.
     * 3. 운용 시점 인프라 가정 & 블랙박스 상호작용 (피드백 3): 외부 환경(특히 외부 물리/조직 시스템)은
     *    운용 시점의 인프라 가정에 기반하며, SoI는 내부 구조를 숨긴 채 경계 포트 입출력으로만 상호작용함.
     */

    private import ScalarValues::*;

    // =========================================================================
    // 1. 임무 정의(Mission Triad) 및 운용효과성측정지표(MOE) 스캐폴드
    //    (교수님 피드백 1, 2 & 가이드 2장 반영)
    // =========================================================================
    package MissionAndMOEScaffold {
        doc
        /* 대상시스템의 존재 목적(What/Why)과 성공 판정 기준(MOE)을 정의하는 추상 계층 */

        // [교수님 피드백 2] 목표값(Target), 한계값(Threshold), 단위(Unit)를 갖춘 MOE 구조체
        attribute def MOEMetric {
            doc
            /* 운용효과성측정지표(Measure of Effectiveness) 표준 템플릿.
             * 수치 자체의 엄밀성보다 '목표값/한계값/단위'의 정형화된 구조 정의가 핵심임.
             */
            attribute metricId : String; // 지표 식별자 (예: "MOE-01")
            attribute metricName : String; // 지표 명칭
            attribute unit : String; // 측정 단위 (예: "%", "분", "건/년", "m")
            attribute thresholdValue : Real; // 한계값 (Threshold: 임무 성공 최소 허용 기준)
            attribute targetValue : Real; // 목표값 (Objective/Target: 지향하는 이상적 기준)
            attribute evaluationCondition : String; // 측정 조건 (예: "ODD 전 구간 운용 기준")
            attribute evaluatedValue : Real [0..1]; // 시나리오/시뮬레이션 수행 후 평가값
        }

        // MOE 달성 여부 판정 제약식 (필요 시 하위 도메인에서 부등호 방향 재정의 가능)
        constraint def MOEAchievementConstraint {
            in moe : MOEMetric;
            moe.evaluatedValue >= moe.thresholdValue
        }

        // [교수님 피드백 1, 2 & 가이드 2장] Mission Triad (What / Why / Validation) 추상 요구사항
        abstract requirement def BaseMissionObjective {
            doc
            /* 모든 도메인의 최상위 임무/사업 목표 요구사항이 상속받아야 할 스캐폴드.
             * 하위 도메인에서 ':>>' (redefines)를 통해 구체적인 텍스트와 MOE를 채워 넣음.
             */
            attribute whatCoreMission : String; // [WHAT] 핵심 임무 선언 (누구를 위해 무엇을 완수하는가)
            attribute whyBusinessObjective : String; // [WHY] 존재 이유 및 사업 목표 (Pain Point 극복 및 창출 효용)
            attribute moeMetrics : MOEMetric [1..*]; // [VALIDATION] 정량적 운용효과성측정지표(MOE) 목록
        }
    } // package MissionAndMOEScaffold

    // =========================================================================
    // 2. 운용 시점(Timeframe) 인프라 가정 및 운용 장소(ODD) 그룹핑 스캐폴드
    //    (교수님 피드백 3 & 가이드 3장, 4장 반영)
    // =========================================================================
    package TimeframeAndODDScaffold {
        doc
        /* [교수님 피드백 3] 블랙박스 SoI가 상호작용할 외부 환경에 대한 가정은
         * 곧 '운용 시점의 인프라·법규·외란에 대한 가정'임을 명시하는 스캐폴드.
         */

        // [가이드 3장 & 피드백 3] 운용 시점 및 4대 기술·제도·인프라 전제 조건
        abstract constraint def BaseTimeframeInfrastructureAssumption {
            doc
            /* 각 워킹그룹이 합의한 운용 시점(예: 2030~2040년)의 4대 외부 환경 전제 조건.
             * 여기서 가정한 인프라와 법규가 3장(BlackBoxContextScaffold)의 외부 액터로 실체화됨.
             */
            attribute targetTimeframe : String; // 운용 시점 (단순 연도가 아닌 기술/제도 앵커)
            attribute externalInfraMaturity : String [1..*]; // ① 외부 인프라 성숙도 (통신/전력/관제 설비 전제)
            attribute regulatoryEnvironment : String [1..*]; // ② 법규 및 제도 환경 (인허가, 무인화 법적 책임 기준)
            attribute threatAndDisturbanceLevel : String [1..*]; // ③ 위협 및 외란 수준 (물리적·전자기적·기상 위협)
            attribute lifecycleEvolutionTrait : String [1..*]; // ④ 수명주기 진화 특성 (배포 후 업그레이드/유지보수 체계)
        }

        // [가이드 4장 & Gate 4] 물리적·기능적 부하(Stress) 동질성 기반 3대 장소 그룹핑
        enum def ODDStressGroupKind {
            GroupA_Nominal; // 그룹 A (공칭/정규 환경): 외부 인프라 완비, 불확실성이 통제된 기본 환경
            GroupB_ComplexHighStress; // 그룹 B (복합/고부하 환경): 이종 객체 혼재 및 시스템 판단 부하 극대화 구역
            GroupC_ExtremeDegraded; // 그룹 C (극한/기능저하 환경): 인프라 음영·극한 외란으로 최소위험상태(MRC) 전이 요구 구역
        }

        part def OperationalLocationGroup {
            attribute groupKind : ODDStressGroupKind;
            attribute representativeLocations : String [1..*]; // 해당 그룹을 대표하는 추상화된 공간 예시
            attribute stressCharacteristics : String; // 시스템에 가해지는 물리적·기능적 부하 특성
        }
    } // package TimeframeAndODDScaffold

    // =========================================================================
    // 3. 블랙박스 대상시스템(SoI) 및 3대 외부 상호작용 환경 스캐폴드
    //    (교수님 피드백 3 & 가이드 1장, 5장, Gate 1·2·5 반영)
    // =========================================================================
    package BlackBoxContextScaffold {
        private import TimeframeAndODDScaffold::*;

        doc
        /* [원칙 1: Solution-Free & Gate 2: Black-Box]
         * 대상시스템(SoI) 내부의 HW/SW 컴포넌트, 제어기, 내부 버스, 알고리즘을 일절 정의하지 않고,
         * 경계면의 포트(Port)와 입출력 교환 아이템(Item)만 정의하는 스캐폴드.
         */

        // ---- 3.1 경계면 입출력 교환 아이템 (Exchange Items) 추상 계층 ----
        abstract item def BaseBoundaryExchangeItem {
            attribute itemDescription : String [0..1];
        }

        // (1) 외부 조직 ↔ SoI 교환 아이템
        item def OrganizationalCommandItem :> BaseBoundaryExchangeItem {
            doc
            /* 외부 조직 -> SoI: 임무 지령, 운용 허가 토큰, 비상 통제 명령 등 */
        }
        item def OrganizationalReportItem :> BaseBoundaryExchangeItem {
            doc
            /* SoI -> 외부 조직: 운용 상태 보고, 임무 결과 로그, 긴급 구난(eCall 등) 요청 */
        }

        // (2) 외부 물리 시스템(인프라) ↔ SoI 교환 아이템
        item def PhysicalInfraInputItem :> BaseBoundaryExchangeItem {
            doc
            /* 외부 물리 인프라 -> SoI: 외부 신호/통신 패킷, 측위 보정, 충전 전력/에너지, 물리적 도킹 등 */
        }
        item def PhysicalInfraOutputItem :> BaseBoundaryExchangeItem {
            doc
            /* SoI -> 외부 물리 인프라: 상태 브로드캐스트 패킷, 인프라 요청 신호, 물리적 작업 출력 등 */
        }

        // (3) 외부 자연 환경 → SoI 외란 아이템 [가이드 5장: 외란 아이템화 규칙]
        item def EnvironmentalDisturbanceItem :> BaseBoundaryExchangeItem {
            doc
            /* 자연 환경과의 상호작용을 문학적 묘사가 아닌 정형 데이터/물리 타입으로 모델링하기 위한 아이템.
             * 외부 자연 환경의 출력 타입과 SoI의 입력 타입 간 대칭성(Type Symmetry)을 보장함.
             */
            attribute disturbanceCategory : String; // 예: "노면 마찰 저하", "폭우 센서 감쇠", "태양풍 방사선"
            attribute parameterName : String; // 예: "마찰계수(mu)", "감쇠율", "열하중"
            attribute parameterValue : Real [0..1]; // 외란 크기(선택)
            attribute parameterUnit : String [0..1]; // 외란 단위(선택)
        }

        // ---- 3.2 포트 대칭성(Port & Type Symmetry)을 보장하는 표준 경계 포트 ----
        // SoI 측 포트 정의 (외부 액터는 '~' 켤레 포트(Conjugated Port)를 사용하여 완벽한 대칭성 유지)
        port def SoI_OrganizationBoundaryPort {
            in item commandIn : OrganizationalCommandItem [0..*];
            out item reportOut : OrganizationalReportItem [0..*];
        }

        port def SoI_PhysicalInfraBoundaryPort {
            in item infraIn : PhysicalInfraInputItem [0..*];
            out item infraOut : PhysicalInfraOutputItem [0..*];
        }

        port def SoI_NaturalEnvBoundaryPort {
            in item disturbanceIn : EnvironmentalDisturbanceItem [0..*];
        }

        // ---- 3.3 3대 외부 상호작용 액터 추상 정의 (조직 / 물리 인프라 / 자연 환경) ----
        abstract part def BaseExternalOrganization {
            doc
            /* ① 외부 조직 (Org): 관제, 법규/인증, 소방/구난 등 인간·사회적 조직 엔티티 */
            attribute orgRole : String;
            // '~' (Conjugation)를 통해 SoI_OrganizationBoundaryPort와 입출력 방향이 정확히 대칭됨
            port orgPort : ~SoI_OrganizationBoundaryPort;
        }

        abstract part def BaseExternalPhysicalSystem {
            doc
            /* ② 외부 물리 시스템 (Physical): [교수님 피드백 3] 운용 시점의 인프라 가정이 실체화된
             * 외부 통신망, 기지국, 충전/보급 설비, 타 시스템 등 인공 물리 엔티티.
             */
            attribute infraSystemRole : String;
            port physicalPort : ~SoI_PhysicalInfraBoundaryPort;
        }

        abstract part def BaseExternalNaturalEnvironment {
            doc
            /* ③ 외부 자연 환경 (Nature): 통제 불가능하며 시스템에 물리적·환경적 외란을 유입하는 자연계 */
            attribute environmentFactor : String;
            port naturePort : ~SoI_NaturalEnvBoundaryPort;
        }

        // ---- 3.4 블랙박스 대상시스템 (Target SoI: Whole-System & Black-Box) ----
        abstract part def BaseBlackBoxSoI {
            doc
            /* [Gate 1: Whole-System & Gate 2: Black-Box]
             * 독립적으로 임무를 완수하는 최상위 전체 시스템이며, 내부 구성요소(Part)를 갖지 않고
             * 오직 3대 외부 환경과 상호작용하는 경계 포트(Port)만 노출함.
             */
            attribute soiName : String;
            port orgBoundaryPort : SoI_OrganizationBoundaryPort [1..*];
            port physicalBoundaryPort : SoI_PhysicalInfraBoundaryPort [1..*];
            port natureBoundaryPort : SoI_NaturalEnvBoundaryPort [1..*];
        }

        // ---- 3.5 블랙박스 운용 컨텍스트 스캐폴드 (Gate 3, 4, 5 내장) ----
        abstract part def BaseBlackBoxOperationalContext {
            doc
            /* 운용 시점 전제 조건, 3대 ODD 장소 그룹, 블랙박스 SoI, 그리고
             * 각 2개 이상의 3대 외부 환경 액터(Gate 5)를 통합하는 컨텍스트 템플릿.
             */
            // [Gate 3] 운용 시점의 인프라·법규·외란 전제 조건 강제
            assert constraint timeframeAssumption : BaseTimeframeInfrastructureAssumption;

            // [Gate 4] 3개 대표 장소 그룹(Group A, B, C) 포함
            part oddGroupA : OperationalLocationGroup {
                :>> groupKind = ODDStressGroupKind::GroupA_Nominal;
            }
            part oddGroupB : OperationalLocationGroup {
                :>> groupKind = ODDStressGroupKind::GroupB_ComplexHighStress;
            }
            part oddGroupC : OperationalLocationGroup {
                :>> groupKind = ODDStressGroupKind::GroupC_ExtremeDegraded;
            }

            // [Gate 1 & 2] 블랙박스 대상시스템(SoI)
            part targetSoI : BaseBlackBoxSoI;

            // [Gate 5: 3대 범위 완전성] 외부 조직, 외부 물리 시스템, 외부 자연 환경 각 2개 이상 식별 강제 ([2..*])
            part externalOrganizations : BaseExternalOrganization [2..*];
            part externalPhysicalSystems : BaseExternalPhysicalSystem [2..*];
            part externalNaturalEnvironments : BaseExternalNaturalEnvironment [2..*];
        }
    } // package BlackBoxContextScaffold

    // 4. 블랙박스 운용 시나리오(ConOps) 및 5-Tuple 명세 스캐폴드
    //    (교수님 피드백 1, 3 & 가이드 6장, Gate 6 반영)
    // =========================================================================
    package BlackBoxConOpsScaffold {
        private import MissionAndMOEScaffold::*;
        private import TimeframeAndODDScaffold::*;
        private import BlackBoxContextScaffold::*;

        doc
        /* [교수님 피드백 1 & 3]
         * 블랙박스 대상시스템이 외부 환경(조직/물리/자연)과 상호작용하며
         * 공칭 운용 -> 외란 적응 -> 비상 폴백(MRC)을 거쳐 존재 목적(임무/MOE)을 달성하는 동적 시나리오 스캐폴드.
         */

        // 대상시스템의 임무 상태 전이(Mission State Transition) 표준 상태 정의
        enum def MissionStateKind {
            StandbyAndInitialized; // 임무 대기 및 초기 인가 상태
            NominalMissionExecuting; // 공칭(Group A) 및 복합(Group B) 환경 정상 임무 수행 상태
            DisturbanceAdapting; // 자연 외란 유입에 적응하여 성능을 유지/조정하는 상태
            MinimalRiskCondition_MRC; // 극한 환경(Group C) 또는 한계 초과 시 최소위험상태(MRC) 안전 폴백 상태
            MissionObjectiveAchieved; // 최종 임무 완수 및 사업 목표(MOE) 달성 상태
        }

        // [가이드 6장] 표준 5-Tuple 시나리오 스텝 추상 액션 정의
        abstract action def BaseScenarioStep_5Tuple {
            doc
            /* 모든 시나리오 스텝은 아래 5-Tuple 요소를 반드시 포함해야 함:
             * 1. [Step N | 장소 그룹] : stepNumber, activeODDGroup
             * 2. [상호작용 액터 (분류)] : interactingOrg / interactingPhysical / interactingNature (Gate 6 추적성)
             * 3. [교환 아이템 (방향)] : inboundItem (Actor -> SoI), outboundItem (SoI -> Actor)
             * 4. [SoI 블랙박스 반응] : soiBlackBoxReaction (내부 구현 배제, 시스템 레벨 의사결정)
             * 5. [임무 상태 전이] : transitionedState (달성된 시스템 임무 상태)
             */
            attribute stepNumber : Integer;
            attribute activeODDGroup : ODDStressGroupKind; // 1. 장소 그룹 (A/B/C)

            // 2. 상호작용 외부 액터 참조 (정의된 모든 액터가 최소 1회 이상 참조되어야 함 - Gate 6)
            ref interactingOrg : BaseExternalOrganization [0..*];
            ref interactingPhysical : BaseExternalPhysicalSystem [0..*];
            ref interactingNature : BaseExternalNaturalEnvironment [0..*];

            // 3. 경계면 교환 아이템 (방향성 명시)
            in item inboundToSoI : BaseBoundaryExchangeItem [0..*]; // [외부 환경 -> SoI] 입력/외란
            out item outboundFromSoI : BaseBoundaryExchangeItem [0..*]; // [SoI -> 외부 환경] 출력/보고

            // 4. SoI 블랙박스 반응 (Solution-Free: 내부 제어기/SW 언급 없이 경계면 반응 기술)
            attribute soiBlackBoxReaction : String;

            // 5. 임무 상태 전이 결과
            attribute transitionedState : MissionStateKind;
        }

        // [가이드 6장 & 피드백 1] 전 과정 블랙박스 운용 시나리오 (ConOps) 추상 액션
        abstract action def BaseBlackBoxConOpsScenario {
            doc
            /* 해피 패스(공칭)뿐 아니라 복합 환경 상호작용, 자연 외란 적응, 비상 폴백(MRC),
             * 그리고 최종 임무/MOE 달성 확인 국면을 시간 순서(Time-ordered sequence)로 통합한 스캐폴드.
             */
            // 국면 1: 공칭 환경(Group A) 임무 개시 및 외부 조직/인프라 상호작용
            action step1_NominalInitiation : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupA_Nominal;
                :>> transitionedState = MissionStateKind::NominalMissionExecuting;
            }

            // 국면 2: 복합/고부하 환경(Group B) 외부 물리 인프라 상호작용
            action step2_ComplexInfraInteraction : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupB_ComplexHighStress;
                :>> transitionedState = MissionStateKind::NominalMissionExecuting;
            }

            // 국면 3: 외부 자연 환경 외란(Disturbance) 유입 및 블랙박스 적응 제어
            action step3_NaturalDisturbanceAdaptation : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupB_ComplexHighStress;
                :>> transitionedState = MissionStateKind::DisturbanceAdapting;
            }

            // 국면 4: 극한/기능저하 환경(Group C) 진입 시 최소위험상태(MRC) 비상 폴백 및 조직 보고
            action step4_ExtremeDegradedMRCFallback : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupC_ExtremeDegraded;
                :>> transitionedState = MissionStateKind::MinimalRiskCondition_MRC;
            }

            // 국면 5: 임무 완수 보고 및 존재 목적(임무/사업 목표 및 MOE) 달성 완료
            action step5_MissionCompletionAndMOEReport : BaseScenarioStep_5Tuple {
                :>> activeODDGroup = ODDStressGroupKind::GroupA_Nominal;
                :>> transitionedState = MissionStateKind::MissionObjectiveAchieved;
            }

            // 시계열 제어 흐름 (하위 도메인에서 세부 스텝 추가/분기 가능)
            first step1_NominalInitiation;
            then step2_ComplexInfraInteraction;
            then step3_NaturalDisturbanceAdaptation;
            then step4_ExtremeDegradedMRCFallback;
            then step5_MissionCompletionAndMOEReport;
        }

             * `constraintType` is a required field that must be set to the type of the constraint. Currently supported types are:
             * - `plain`: The constraint is regular constraint.
             * - `assumed`: The constraint is an assumed constraint.
             * - `asserted`: The constraint is an asserted constraint.
             * - `required`: The constraint is a required constraint.
             *
             * `constraintLanguage` is a required field that must be set to the language of the constraint.
             *
             * **Editing**
             *
             * This is an explicit element -> **supports** editing through the UI.
             *
             * Similar to `attributeValue`, the table view requires non-empty `constraintName` due to inheritence
             */
            attribute constraintName default null;
            attribute constraintType default ConstraintType::plain;
            attribute constraintLanguage default "English";
        }
    }

        view def <MV> MatrixView {
            view rowView : ContentView [1] {
                doc
                /* Inside this view use `expose` and `filter` to specify what you want to see
                 * as rows in the matrix.
                 */
                attribute representation : Rep default Rep::name;
            }
            view columnView : ContentView [1] {
                doc
                /* Inside this view use `expose` and `filter` to specify what you want to see
                 * as columns in the matrix.
                 */
                attribute representation : Rep default Rep::name;
                attribute verticalColumnNames default false;
            }
            view cellView : ContentView [1] {
                doc
                /* Inside this view use `expose`, `filter`, and `direction` to specify what you want to see
                 * as cells in the matrix.
                 */
                attribute :>> includeReferenceElements default true;
                attribute direction : Dir default Dir::undirected;
                attribute showWeakTraceability default false;
                attribute portsAreTransparent default false;
                /* If true, connections to ports will be treated as direct connections to the port's owner */
                filter (
                    as SysML::ConnectionUsage hastype SysML::ConnectionUsage or
                    as SysML::AllocationUsage hastype SysML::AllocationUsage or
                    @ RequirementDerivation::derivation
                );
            }
        }

                filter as SysML::Definition istype SysML::Definition;

            }
            view :>> cellView {
                attribute :>> exposeFeaturesAndHeritage = true;
            }
        }

        view def ConnectionMatrixView :> EditableMatrixView {
            view :>> cellView {
                filter (
                    as SysML::ConnectionUsage hastype SysML::ConnectionUsage and
                    not (@ RequirementDerivation::derivation)
                );
            }
        }

        view def AllocationMatrixView :> EditableMatrixView {
            view :>> cellView {
                filter as SysML::AllocationUsage hastype SysML::AllocationUsage;
            }
        }

        view def RequirementMatrixView :> EditableMatrixView {
            view :>> cellView {
                filter @ RequirementDerivation::derivation;
            }
        }
